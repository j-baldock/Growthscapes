# Helper: extract Oct 1 experimental-period rows from one ibm_long
# Filtering before bind_rows() keeps only ~1/365th of each data frame.
extract_oct1_exp <- function(ibm_long, scen_key) {
  ibm_long |>
    filter(date >= exp_start_date,
           month(date) == 10, day(date) == 1,
           survived == 1) |>
    mutate(scen_key  = scen_key,
           year      = year(date),
           age_class = floor(age))
}

# Helper: compute per-year biomass accrual by patch (experimental period only).
# Summarises to n_years × 2 rows per scenario before combining.
compute_patch_growth <- function(ibm_long, scen_key) {
  ibm_long |>
    filter(date >= exp_start_date, !is.na(ggd)) |>
    mutate(daily_growth_g = weight * ggd, year = year(date)) |>
    group_by(year, patch) |>
    summarise(total_g = sum(daily_growth_g, na.rm = TRUE), .groups = "drop") |>
    mutate(scen_key = scen_key)
}

Compute unfished adult abundance (the portion of the population eligible to spawn, >200g) on a specified day of year and export as a .csv. This is used to derive the target harvest abundance for scenarios with variable exploitation rates.

Code 
# ── Unfished adult abundance on harvest day ───────────────────────────────────
# Computes mean annual count of fish >= 200 g (the harvest eligibility floor,
# w_min in fncMaturitySize) on a specified day-of-year across all experimental-
# period years. Used to parameterise n_unfished in harvest simulations.
#
# Census is taken on harvest_doy to match the demographic snapshot seen by the
# harvest module — accounting for seasonal weight cycles and post-spawn weight
# loss that can shift fish across the 200 g threshold relative to the Oct 1
# census used elsewhere.

harvest_doy <- 91L  # day-of-year of harvest pulse (change to match harvest_doy in params)

adult_sum <- purrr::map_dfr(ibm_list_mech, function(scen) {
  scen$df |>
    filter(
      date       >= exp_start_date,
      yday(date) == harvest_doy,
      weight     >= 200,
      survived   == 1
    ) |>
    mutate(year = year(date)) |>
    count(year) |>
    summarise(mean_n = mean(n)) |>
    mutate(
      hab_type  = scen$ht,
      prop_cold = scen$pc,
      A_cold    = scen$pc * 2,
      A_warm    = if (scen$ht == "Cold only") 0 else (1 - scen$pc) * 2
    )
}) |>
  left_join(scen_meta |> select(prop_cold, label), by = "prop_cold") |>
  select(hab_type, label, prop_cold, A_cold, A_warm, mean_n) |>
  mutate(
    hab_type = factor(hab_type, levels = c("Cold only", "Cold + Warm", "Cold + Warm (high P)")),
    label    = factor(label,    levels = scen_meta$label)
  ) |>
  arrange(hab_type, desc(prop_cold))

write_csv(adult_sum, "adult_sum.csv")

# Birth events only — filtered and immediately aggregated to doy counts to
# avoid carrying 18 M individual rows forward into geom_density()
f_spawn_dates <- function(df) {
  df |>
    filter(!is.na(parent_pid),
           dayofsim == birth_dayofsim,
           date >= exp_start_date) |>
    mutate(doy = yday(date)) |>
    count(doy, name = "n")
}

# Pre-aggregate: one row per (scenario, cold proportion, doy)
spawn_date_df <- bind_rows(lapply(ibm_list_mech, \(s) {
  f_spawn_dates(s$df) |>
    mutate(prop_cold = s$pc,

    label    = factor(label, levels = scen_meta$label),

  group_by(hab_type, label) |>
