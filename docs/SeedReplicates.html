12.1 Setup

Load each result .rds object and bind directly into a list. Using a helper function ensures each res object is freed from memory as soon as ibm_long is extracted, avoiding duplicate references that would force R to copy the data frames.

# Helper: read one scenario, extract ibm_long and spawn_log, and free res before returning
load_scen <- function(path, repl, pc, ht, pcmax_cold, pcmax_warm) {
  res <- readRDS(path)

       repl = repl, pc = pc, ht = ht, pcmax_cold = pcmax_cold, pcmax_warm = pcmax_warm)
}

# (ibm_long is NULL, has no rows, or lacks a date column)
ibm_is_empty <- function(ibm_long) {
  is.null(ibm_long) || nrow(ibm_long) == 0 || !"date" %in% names(ibm_long)

# Helper: extract Oct 1 experimental-period rows from one ibm_long
# Filtering before bind_rows() keeps only ~1/365th of each data frame.
extract_oct1_exp <- function(ibm_long, scen_key) {
  if (ibm_is_empty(ibm_long)) return(tibble())
  ibm_long |>
    filter(date >= exp_start_date,
           month(date) == 10, day(date) == 1,
           survived == 1) |>
    mutate(scen_key  = scen_key,
           year      = year(date),
           age_class = floor(age),
           peak_weight = weight / condition)
}

# Helper: compute per-year biomass accrual by patch (experimental period only).
# Summarises to n_years × 2 rows per scenario before combining.
compute_patch_growth <- function(ibm_long, scen_key) {
  if (ibm_is_empty(ibm_long)) return(tibble())
  ibm_long |>
    filter(date >= exp_start_date, !is.na(ggd)) |>
    mutate(daily_growth_g = weight * ggd, year = year(date)) |>
    group_by(year, patch) |>
    summarise(total_g = sum(daily_growth_g, na.rm = TRUE), .groups = "drop") |>
    mutate(scen_key = scen_key)
}

# ── Oct 1 census: experimental period, all hab types, all scenarios ───────────
exp_oct1 <- purrr::map_dfr(ibm_list_mech, function(scen) {

  )

# ── Annual spawner totals per scenario × hab_type × year ─────────────────────
# Each parent may appear multiple times per year (repeat spawning attempts);
# average weight within (year, parent) before summing to avoid double-counting mass.
exp_spawn_yr <- purrr::map_dfr(ibm_list_mech, function(scen) {
  scen$spawn_log |>
    mutate(date = base_params$mindate + (dayofsim - 1L),
           yr   = year(date)) |>
    filter(date >= exp_start_date) |>
    group_by(yr, parent_pid) |>
    summarise(parent_wt = mean(weight), .groups = "drop") |>
    group_by(yr) |>
    summarise(n_spawners    = n_distinct(parent_pid),
              biom_spawn_kg = sum(parent_wt) / 1000,
              .groups = "drop") |>
    mutate(hab_type = scen$ht, prop_cold = scen$pc, repl = scen$repl)
}) |>
  left_join(scen_meta, by = "prop_cold") |>
  mutate(
    label    = factor(label,    levels = scen_meta$label),
    hab_type = factor(hab_type, levels = c("Cold only", "Cold + Warm", "Cold + Warm (high P)"))
  )

12.2 Key metrics

Same metrics as the “Summary of key metrics” section of CompareOutcomes.qmd, shown as the distribution of annual values (experimental period) for each replicate. Maximum age and size are the annual 99th percentiles in the Oct 1 census. Age at first reproduction is the annual mean among parents whose first spawn occurred that year. Spawning participation is the number of spawners in a year divided by the Oct 1 census of the previous year.

Code 
first_repro <- purrr::map_dfr(ibm_list_mech, function(scen) {
  sl <- scen$spawn_log
  births <- sl |>
    distinct(dayofsim, offspring_pid_start, offspring_pid_end) |>
    rename(birth_day = dayofsim)

  sl |>
    mutate(date = base_params$mindate + (dayofsim - 1L)) |>
    left_join(births, by = join_by(between(parent_pid, offspring_pid_start, offspring_pid_end))) |>
    filter(!is.na(birth_day)) |>
    arrange(parent_pid, dayofsim) |>
    distinct(parent_pid, .keep_all = TRUE) |>
    filter(date >= exp_start_date) |>
    transmute(parent_pid,
              year      = year(date),
              age_first = (dayofsim - birth_day) / 365,
              hab_type  = scen$ht,
              prop_cold = scen$pc,
              repl      = scen$repl)
}) |>
  left_join(scen_meta, by = "prop_cold") |>
  mutate(label    = factor(label,    levels = scen_meta$label),
         hab_type = factor(hab_type, levels = levels(exp_oct1$hab_type)))

annual_max <- exp_oct1 |>
  group_by(hab_type, repl, year) |>
  summarise(max_age = quantile(age, 0.99),
            max_wt  = quantile(weight, 0.99),
            .groups = "drop")

annual_repro <- first_repro |>
  group_by(hab_type, repl, year) |>
  summarise(age_first = mean(age_first), .groups = "drop")

# Spawners in year y relative to the Oct 1 census of year y - 1
annual_part <- exp_spawn_yr |>
  inner_join(exp_pop_yr |> transmute(hab_type, label, repl, yr = year + 1L, n_census = n),
             by = c("hab_type", "label", "repl", "yr")) |>
  transmute(hab_type, repl, year = yr, participation = n_spawners / n_census)

annual_all <- annual_max |>
  full_join(annual_repro, by = c("hab_type", "repl", "year")) |>
  full_join(annual_part,  by = c("hab_type", "repl", "year")) |>
  mutate(repl = factor(repl))

12.7 Warm habitat contribution

The relative importance of warm habitat is estimated with the counterfactual approach from WarmContribution.qmd: each Cold + Warm run is compared with the Cold only run at the same cold proportion and replicate (replicates share a seed across scenarios), so that:

\[\text{benefit} = \underbrace{N_{warm}}_{\text{direct}} + \underbrace{(N_{cold} - N_{cold,\,\text{cold only}})}_{\text{indirect}}\]

where \(N\) is signed net growth (g). Years are paired between scenarios, and the first and last calendar years are dropped because the simulation starts and ends on May 1. Importance is expressed as a fraction of annual Cold + Warm net production, so seasonal fractions sum to the annual fraction.

Code 
cf_seas <- c("Winter", "Spring", "Summer", "Fall")
yr_drop <- range(patch_growth$year)
cf_comp_cols <- c("Warm patch net growth (direct)" = unname(hab_cols["Cold + Warm"]),
                  "Change in cold-patch growth vs Cold only (indirect)" = unname(hab_cols["Cold only"]))

theme_cf <- theme_bw(base_size = 12) +
  theme(panel.grid = element_blank(), axis.text = element_text(color = "black"))

# Seasonal net growth by patch (summarised within each run before combining)
compute_patch_growth_seasonal <- function(ibm_long) {
  if (ibm_is_empty(ibm_long)) return(tibble())
  ibm_long |>
    filter(date >= exp_start_date, !is.na(ggd)) |>
    mutate(daily_growth_g = weight * ggd,
           year   = year(date),
           season = case_when(month(date) %in% c(12, 1, 2)  ~ "Winter",
                              month(date) %in% c(3, 4, 5)   ~ "Spring",
                              month(date) %in% c(6, 7, 8)   ~ "Summer",
                              month(date) %in% c(9, 10, 11) ~ "Fall"),
           season = factor(season, levels = cf_seas)) |>
    group_by(year, season, patch) |>
    summarise(total_g = sum(daily_growth_g, na.rm = TRUE), .groups = "drop")
}

patch_growth_seasonal <- purrr::map_dfr(ibm_list_mech, function(scen) {
  compute_patch_growth_seasonal(scen$df) |>
    mutate(hab_type = scen$ht, repl = scen$repl)
}) |>
  filter(!year %in% yr_drop)

# Annual counterfactual
cold_only_annual <- patch_growth |>
  filter(hab_type == "Cold only", patch == "cold") |>
  select(repl, year, C_co = total_g)

cf_annual <- patch_growth |>
  filter(hab_type == "Cold + Warm", !year %in% yr_drop) |>
  select(repl, year, patch, total_g) |>
  pivot_wider(names_from = patch, values_from = total_g, values_fill = 0) |>
  left_join(cold_only_annual, by = c("repl", "year")) |>
  mutate(direct   = warm,
         indirect = cold - C_co,
         benefit  = direct + indirect,
         I_cf     = benefit / (warm + cold),
         repl     = factor(repl))

# Seasonal counterfactual
cold_only_seasonal <- patch_growth_seasonal |>
  filter(hab_type == "Cold only", patch == "cold") |>
  select(repl, year, season, C_co = total_g)

cf_seasonal <- patch_growth_seasonal |>
  filter(hab_type == "Cold + Warm") |>
  select(repl, year, season, patch, total_g) |>
  pivot_wider(names_from = patch, values_from = total_g, values_fill = 0) |>
  left_join(cold_only_seasonal, by = c("repl", "year", "season")) |>
  mutate(direct   = warm,
         indirect = cold - C_co,
         benefit  = direct + indirect) |>
  group_by(repl, year) |>
  mutate(annual_cw = sum(warm + cold)) |>
  ungroup() |>
  mutate(I_seas = benefit / annual_cw,
         repl   = factor(repl))

## Warm habitat contribution

The relative importance of warm habitat is estimated with the counterfactual approach from WarmContribution.qmd: each Cold + Warm run is compared with the Cold only run at the same cold proportion and replicate (replicates share a seed across scenarios), so that:

$$\text{benefit} = \underbrace{N_{warm}}_{\text{direct}} + \underbrace{(N_{cold} - N_{cold,\,\text{cold only}})}_{\text{indirect}}$$

where $N$ is signed net growth (g). Years are paired between scenarios, and the first and last calendar years are dropped because the simulation starts and ends on May 1. Importance is expressed as a fraction of annual Cold + Warm net production, so seasonal fractions sum to the annual fraction.

```{r}
#| cache: false

cf_seas <- c("Winter", "Spring", "Summer", "Fall")
yr_drop <- range(patch_growth$year)
cf_comp_cols <- c("Warm patch net growth (direct)" = unname(hab_cols["Cold + Warm"]),
                  "Change in cold-patch growth vs Cold only (indirect)" = unname(hab_cols["Cold only"]))

theme_cf <- theme_bw(base_size = 12) +
  theme(panel.grid = element_blank(), axis.text = element_text(color = "black"))

# Seasonal net growth by patch (summarised within each run before combining)
compute_patch_growth_seasonal <- function(ibm_long) {
  if (ibm_is_empty(ibm_long)) return(tibble())
  ibm_long |>
    filter(date >= exp_start_date, !is.na(ggd)) |>
    mutate(daily_growth_g = weight * ggd,
           year   = year(date),
           season = case_when(month(date) %in% c(12, 1, 2)  ~ "Winter",
                              month(date) %in% c(3, 4, 5)   ~ "Spring",
                              month(date) %in% c(6, 7, 8)   ~ "Summer",
                              month(date) %in% c(9, 10, 11) ~ "Fall"),
           season = factor(season, levels = cf_seas)) |>
    group_by(year, season, patch) |>
    summarise(total_g = sum(daily_growth_g, na.rm = TRUE), .groups = "drop")
}

patch_growth_seasonal <- purrr::map_dfr(ibm_list_mech, function(scen) {
  compute_patch_growth_seasonal(scen$df) |>
    mutate(hab_type = scen$ht, repl = scen$repl)
}) |>
  filter(!year %in% yr_drop)

# Annual counterfactual
cold_only_annual <- patch_growth |>
  filter(hab_type == "Cold only", patch == "cold") |>
  select(repl, year, C_co = total_g)

cf_annual <- patch_growth |>
  filter(hab_type == "Cold + Warm", !year %in% yr_drop) |>
  select(repl, year, patch, total_g) |>
  pivot_wider(names_from = patch, values_from = total_g, values_fill = 0) |>
  left_join(cold_only_annual, by = c("repl", "year")) |>
  mutate(direct   = warm,
         indirect = cold - C_co,
         benefit  = direct + indirect,
         I_cf     = benefit / (warm + cold),
         repl     = factor(repl))

# Seasonal counterfactual
cold_only_seasonal <- patch_growth_seasonal |>
  filter(hab_type == "Cold only", patch == "cold") |>
  select(repl, year, season, C_co = total_g)

cf_seasonal <- patch_growth_seasonal |>
  filter(hab_type == "Cold + Warm") |>
  select(repl, year, season, patch, total_g) |>
  pivot_wider(names_from = patch, values_from = total_g, values_fill = 0) |>
  left_join(cold_only_seasonal, by = c("repl", "year", "season")) |>
  mutate(direct   = warm,
         indirect = cold - C_co,
         benefit  = direct + indirect) |>
  group_by(repl, year) |>
  mutate(annual_cw = sum(warm + cold)) |>
  ungroup() |>
  mutate(I_seas = benefit / annual_cw,
         repl   = factor(repl))
```
