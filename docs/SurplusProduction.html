13.1 Setup

Load relevant .rds objects and bind directly into a list. Using a helper function ensures each res object is freed from memory as soon as ibm_long is extracted, avoiding duplicate references that would force R to copy the data frames.

sim_params     <- res$params
habitat_df     <- res$habitat_df
exp_start_date <- sim_params$mindate + years(sim_params$nyears_burnin)
rm(res)

# Helper: read one scenario, extract ibm_long, and free res before returning
load_scen <- function(path, ht, pc, ac, aw, hr, pcmax_cold, pcmax_warm) {
  res <- readRDS(path)
  list(df = res$ibm_long, harvest = res$harvest_log, 
       ht = ht, pc = pc, ac = ac, aw = aw, hr = hr, pcmax_cold = pcmax_cold, pcmax_warm = pcmax_warm)
}

13.3 Recruitment

The number of fry produced is a function of adult size and abundance (fecundity), which are driven by warm and cold habitat availability. Importantly, when only cold habitat is available, the number of fry produced declines ~linearly with cold habitat availability. But when warm habitat is also available, considerably more fry are produced at a given proportion cold (except for 95% cold); i.e.,

Code 
# ── Annual age-0 recruitment time series ─────────────────────────────────────
# Zero-fill missing years so post-crash scenarios contribute zeros for the
# remainder of the experimental period rather than being silently excluded.
all_exp_years <- tibble(
  yr = seq(year(exp_start_date),
           year(exp_start_date) + sim_params$nyears_experiment - 1L)
)

get_annual_age0 <- function(scen) {
  scen$df |>
    filter(date >= exp_start_date,
           month(date) == 11, day(date) == 1,
           survived == 1, floor(age) == 0) |>
    count(yr = year(date), name = "n_age0") |>
    right_join(all_exp_years, by = "yr") |>
    replace_na(list(n_age0 = 0L)) |>
    mutate(hab_type = scen$ht,
           harvest_rate = factor(scen$hr,
                                 levels = hr_levels_mech,
                                 labels = hr_labels_mech))
}

age0_ts <- purrr::map_dfr(ibm_list_mech, get_annual_age0)


# ── End-vs-start recruitment ratio ───────────────────────────────────────────
# Collapses each trajectory to a single scalar: mean n_age0 in the last 10
# experimental years divided by the first 10. Stable harvest → ratio ≈ 1;
# slow decline → ratio < 1; crashed population → ratio = 0.
n_exp   <- nrow(all_exp_years)
first10 <- all_exp_years$yr[1:10]
last10  <- all_exp_years$yr[(n_exp - 9):n_exp]

ratio_df <- age0_ts |>
  mutate(period = case_when(
    yr %in% first10 ~ "start",
    yr %in% last10  ~ "end",
    TRUE            ~ NA_character_
  )) |>
  filter(!is.na(period)) |>
  group_by(hab_type, harvest_rate, period) |>
  summarise(mean_n = mean(n_age0), sd_n = sd(n_age0), .groups = "drop") |>
  pivot_wider(names_from = period, values_from = c(mean_n, sd_n)) |>
  mutate(ratio = mean_n_end / mean_n_start) |>
  left_join(
    age0_ts |>
      group_by(hab_type, harvest_rate) |>
      summarise(mean_n_all = mean(n_age0), sd_n_all = sd(n_age0), .groups = "drop"),
    by = c("hab_type", "harvest_rate")
  )

Summarize harvest data

all_exp_years <- tibble(
  year = seq(year(exp_start_date),
           year(exp_start_date) + sim_params$nyears_experiment - 1L)
)

get_annual_harvest <- function(scen) {
  scen$harvest %>%
    group_by(year) %>%
    summarise(n_harvested = n(),
              bm_harvested = sum(weight)/1000,
              mean_wt = mean(weight), 
              sd_wt = sd(weight),
              mean_age = mean(age),
              sd_age = sd(age),
              mean_cond = mean(condition),
              sd_cond = sd(condition)) %>%
    ungroup() %>%
    right_join(all_exp_years, by = "year") %>%
    replace_na(list(n_harvested = 0L,
                    bm_harvested = 0L,
                    mean_wt = NA,
                    sd_wt = NA,
                    mean_age = NA,
                    sd_age = NA,
                    mean_cond = NA,
                    sd_cond = NA)) %>%
    mutate(hab_type = scen$ht,
           harvest_rate = factor(scen$hr,
                                 levels = hr_levels_mech,
                                 labels = hr_labels_mech))
}

get_harvest_traits <- function(scen) {
  scen$harvest %>%
    summarise(mean_wt = mean(weight), 
              sd_wt = sd(weight),
              mean_age = mean(age),
              sd_age = sd(age),
              mean_cond = mean(condition),
              sd_cond = sd(condition)) %>%
    ungroup() %>%
    mutate(hab_type = scen$ht,
           harvest_rate = factor(scen$hr,
                                 levels = hr_levels_mech,
                                 labels = hr_labels_mech))
}

get_annual_age0 <- function(scen) {
  scen$df |>
    filter(date >= exp_start_date,
           month(date) == 11, day(date) == 1,
           survived == 1, floor(age) == 0) |>

## Fry production vs. recruitment
