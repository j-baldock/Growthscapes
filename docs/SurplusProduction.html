16  Surplus Production

Purpose: Compare surplus production and the impacts of “harvest” across habitat scenarios and cold-habitat proportions. Surplus production is estimated directly from the simulated population using a biomass budget (census biomass plus catch; see Surplus production: biomass budget). The fitted stock-recruitment (Ricker) analyses are retained below as a supplement.

Note that we currently only have harvest simulations run for 75% cold proportion.

16.1 Setup

Load relevant .rds objects and bind directly into a list. Most outputs come from the summary objects (ibm_summary, spawn_log, harvest_log). ibm_long is used only inside compute_budget(), which reduces it to an annual biomass budget for the exploitable stock, so the full individual-level data frame is never stored. A helper function frees each res object as soon as it is processed.

Code 
# get experiment start date from base parameters
exp_start_date <- base_params$mindate + years(base_params$nyears_burnin)

# Helper: read one scenario, pull what is needed from the summary objects, and free res.
# ibm_long is never touched, so it is never copied. Extracted objects:
#   oct1_waa   – Oct 1 weight-at-age summary (n, mean, sd) by year x age class,
#                pooled over strategy and patch (ibm_summary is one row per
#                strategy x patch x age_class x month-start)
#   oct1_adult – Oct 1 adults (peak_weight > 200 g): count and biomass (kg) by year
#   age0_nov1  – annual age-0 counts and biomass on Nov 1
#   fry_annual – annual fry birth counts (sum of spawn_log offspring)
#   budget     – annual biomass budget of the exploitable stock (see compute_budget)
#   spawners   – annual spawner counts, size, age, new-entrant fraction and fry (see compute_spawners)
# Requires results generated with the n_adult / biomass_adult_g columns in ibm_summary.

# Annual biomass budget of the exploitable stock (peak_weight >= 200 g, matching
# harvest eligibility), from ibm_long and harvest_log.
#
# Budget year t runs from the census on day-of-year 151 of year t (the last day before
# the harvest window opens on DOY 152, so the census is pre-harvest) to the same census
# in year t + 1. Each fish-day transition d -> d + 1 of a stock member is classified as
#   stay    – survives; weight change is growth
#   die     – natural death on d + 1; weight at death leaves the stock
#   harvest – no ibm_long row on d + 1 and a harvest_log record; harvest weight leaves
# so that, exactly,  B[t+1] - B[t] = G_net + R - M - C  and  SP = B[t+1] - B[t] + C = G_net + R - M.
#   G_gross  – sum of weight x ggd over stock members (growth before spawning losses)
#   L_spawn  – G_gross - G_net: weight lost by stock members other than through death or
#              harvest (spawning); not in the simple SP = G + R - M decomposition
#   R        – biomass of fish at the weight they first reach peak_weight >= 200 g
# peak_weight is reconstructed as the running max of weight within ibm_long, so fish
# already present on the first day lack pre-history; the first years are dropped downstream.
# All biomass in kg.
compute_budget <- function(ibm_long, harvest_log, habitat_df, census_doy = 151L) {
  empty <- tibble(yr = integer(), B = numeric(), C = numeric(), G_net = numeric(),
                  G_gross = numeric(), L_spawn = numeric(), R = numeric(),
                  M = numeric(), B_next = numeric(), SP = numeric())
  if (is.null(ibm_long) || nrow(ibm_long) == 0) return(empty)

  il <- data.table::as.data.table(ibm_long)[, .(pid, dayofsim, weight, ggd, survived)]
  data.table::setorder(il, pid, dayofsim)
  cens <- habitat_df |>
    filter(doy == census_doy, dayofsim >= min(il$dayofsim), dayofsim <= max(il$dayofsim)) |>
    transmute(dayofsim, yr = year(date))
  first_day <- min(il$dayofsim)

  il[, peak := cummax(weight), by = pid]
  il[, `:=`(w_nx   = data.table::shift(weight,   -1L),
            d_nx   = data.table::shift(dayofsim, -1L),
            s_nx   = data.table::shift(survived, -1L),
            pid_nx = data.table::shift(pid,      -1L))]
  il[, nx_ok := !is.na(pid_nx) & pid_nx == pid & d_nx == dayofsim + 1L]
  il[, stock := survived == 1L & peak >= 200]

  # Harvested fish have no row on the harvest day; key their harvest weight to the day before
  hl <- data.table::as.data.table(harvest_log)[, .(pid, dayofsim = dayofsim - 1L, h_w = weight)]
  st <- merge(il[stock == TRUE], hl, by = c("pid", "dayofsim"), all.x = TRUE, sort = FALSE)
  st[, event := data.table::fifelse(nx_ok & s_nx == 1L, "stay",
                data.table::fifelse(nx_ok & s_nx == 0L, "die",
                data.table::fifelse(!is.na(h_w), "harvest", "end")))]
  st <- st[event != "end"]
  st[, w_out := data.table::fifelse(event == "harvest", h_w, w_nx)]
  # Budget year of the flow, indexed by the census that opens it
  st[, t_idx := findInterval(dayofsim, cens$dayofsim)]
  flows <- st[t_idx >= 1L, .(G_net   = sum(w_out - weight),
                             G_gross = sum(weight * ggd),
                             M       = sum(data.table::fifelse(event == "die",     w_out, 0)),
                             C       = sum(data.table::fifelse(event == "harvest", w_out, 0))),
              by = t_idx]

  # Recruits: first alive day at peak >= 200 g (fish present on day 1 are initial stock)
  rec <- unique(il[survived == 1L & peak >= 200 & dayofsim > first_day], by = "pid")
  rec[, t_idx := findInterval(dayofsim - 1L, cens$dayofsim)]
  rec <- rec[t_idx >= 1L, .(R = sum(weight)), by = t_idx]

  bio <- il[dayofsim %in% cens$dayofsim & stock == TRUE, .(B = sum(weight)), by = dayofsim]

  cens |>
    mutate(t_idx = row_number()) |>
    left_join(as_tibble(bio),   by = "dayofsim") |>
    left_join(as_tibble(flows), by = "t_idx") |>
    left_join(as_tibble(rec),   by = "t_idx") |>
    arrange(t_idx) |>
    mutate(across(c(B, G_net, G_gross, M, C, R), \(x) replace_na(x, 0) / 1000),
           B_next  = lead(B),
           L_spawn = G_gross - G_net) |>
    filter(!is.na(B_next)) |>
    mutate(SP = B_next - B + C) |>
    select(yr, B, C, G_net, G_gross, L_spawn, R, M, B_next, SP)
}

# Annual spawner summary from spawn_log, one row per spawning year in the experimental period.
# A spawner is a "new entrant" if it first reached peak_weight >= 200 g (reconstructed as the
# running max of weight in ibm_long) after the most recent harvest-window opening (DOY 152),
# i.e. it was never eligible for harvest before this spawning season. Fish already >= 200 g on
# the first day of ibm_long have unknown crossing dates and count as not new.
#   n_spawners, spawn_biom_kg – distinct parents and their summed weight at spawning
#   mean_wt, mean_age         – per spawning event; age from fish_registry birth dates
#   frac_new                  – fraction of distinct spawners that were new entrants
#   fry                       – sum of n_offspring
compute_spawners <- function(ibm_long, spawn_log, fish_registry, habitat_df, first_date,
                             window_open_doy = 152L) {
  empty <- tibble(yr = integer(), n_spawners = integer(), n_events = integer(),
                  mean_wt = numeric(), mean_age = numeric(), frac_new = numeric(),
                  spawn_biom_kg = numeric(), fry = numeric())
  if (is.null(ibm_long) || nrow(ibm_long) == 0 || nrow(spawn_log) == 0) return(empty)

  il <- data.table::as.data.table(ibm_long)[, .(pid, dayofsim, weight)]
  data.table::setorder(il, pid, dayofsim)
  il[, peak := cummax(weight), by = pid]
  first_day <- min(il$dayofsim)
  cross <- il[peak >= 200, .(cross_day = min(dayofsim)), by = pid]
  cross[cross_day == first_day, cross_day := NA_integer_]
  rm(il)

  win_open <- habitat_df$dayofsim[habitat_df$doy == window_open_doy]
  spawn_log |>
    mutate(date = habitat_df$date[dayofsim], yr = year(date)) |>
    filter(date >= first_date) |>
    left_join(select(fish_registry, pid, birth_dayofsim), by = c("parent_pid" = "pid")) |>
    left_join(as_tibble(cross), by = c("parent_pid" = "pid")) |>
    mutate(prev_open   = win_open[findInterval(dayofsim, win_open)],
           age         = (dayofsim - birth_dayofsim) / 365.25,
           new_entrant = !is.na(cross_day) & cross_day >= prev_open) |>
    group_by(yr) |>
    summarise(n_spawners    = n_distinct(parent_pid),
              n_events      = n(),
              mean_wt       = mean(weight),
              mean_age      = mean(age),
              frac_new      = n_distinct(parent_pid[new_entrant]) / n_distinct(parent_pid),
              spawn_biom_kg = sum(weight[!duplicated(parent_pid)]) / 1000,
              fry           = sum(n_offspring),
              .groups = "drop")
}

load_scen <- function(path, ht, pc, ac, aw, hr, pcmax_cold, pcmax_warm) {
  res <- readRDS(path)
  stopifnot("ibm_summary lacks patch/age_class/n_adult; re-run simulation" =
              all(c("patch", "age_class", "n_adult") %in% names(res$ibm_summary)))

  sm <- res$ibm_summary |>
    filter(date >= exp_start_date)

  sm_oct1 <- sm |> filter(month(date) == 10, day(date) == 1)

  # Pool strategy x patch groups within each age class. Group means are
  # n-weighted (sum of biomass / sum of n); pooled variance combines within-group
  # and between-group sums of squares. sd_weight is NA for single-fish groups;
  # their (n - 1) weight is 0 anyway.
  oct1_waa <- sm_oct1 |>
    filter(n_alive > 0) |>
    mutate(ss_within = (n_alive - 1) * replace_na(sd_weight, 0)^2,
           ss_mean   = n_alive * mean_weight^2) |>
    group_by(yr = year(date), age_class) |>
    summarise(
      n           = sum(n_alive),
      mean_weight = sum(biomass_g) / n,
      sd_weight   = sqrt(pmax(sum(ss_within) + sum(ss_mean) - n * mean_weight^2, 0) /
                           pmax(n - 1, 1)),
      .groups = "drop"
    )

  # Sum across strategy, patch, and age class
  oct1_adult <- sm_oct1 |>
    group_by(yr = year(date)) |>
    summarise(n_oct1    = sum(n_adult),
              biom_oct1 = sum(biomass_adult_g) / 1000,  # g -> kg
              .groups = "drop")

  # Age-0 only; summed across strategy and patch
  age0_nov1 <- sm |>
    filter(month(date) == 11, day(date) == 1, age_class == 0) |>
    group_by(yr = year(date)) |>
    summarise(n_age0 = sum(n_alive), biom_age0 = sum(biomass_g), .groups = "drop")

  fry_annual <- res$spawn_log |>
    mutate(date = res$habitat_df$date[dayofsim]) |>
    filter(date >= exp_start_date) |>
    group_by(yr = year(date)) |>
    summarise(n_fry = sum(n_offspring), .groups = "drop")

  list(oct1_waa = oct1_waa, oct1_adult = oct1_adult,
       age0_nov1 = age0_nov1, fry_annual = fry_annual,
       budget = compute_budget(res$ibm_long, res$harvest_log, res$habitat_df),
       spawners = compute_spawners(res$ibm_long, res$spawn_log, res$fish_registry,
                                   res$habitat_df, exp_start_date),
       harvest = res$harvest_log, spawn_log = res$spawn_log,
       ht = ht, pc = pc, ac = ac, aw = aw, hr = hr,
       pcmax_cold = pcmax_cold, pcmax_warm = pcmax_warm)
}

16.4 Surplus production: biomass budget

Surplus production is the biomass the stock adds in a year beyond what is needed to replace itself, i.e., the yield that could be taken while leaving the stock the same size. Because the IBM tracks every fish, it can be measured directly with a biomass budget, with no stock-recruitment model and no choice of lag:

\[SP_t = B_{t+1} - B_t + C_t\]

where \[B_t\] is the biomass of the exploitable stock at a fixed census date and \[C_t\] is the biomass harvested between censuses. At equilibrium (\[B_{t+1} = B_t\]) surplus production equals catch by definition, which is why the Ricker-based estimates further below track the harvest summaries so closely.

Implementation (see compute_budget() in Setup):

  • Stock: fish with \[peak\_weight \geq 200\] g, matching harvest eligibility. ibm_long has no peak_weight column, so it is reconstructed as the running maximum of each fish’s weight within ibm_long.
  • Census: day of year 151, the last day before the harvest window opens (DOY 152), every year. ibm_long is recorded after that day’s harvest, so a census on DOY 152 would already be missing the first day’s catch; DOY 151 keeps the budget closed.
  • Catch: sum of harvest_log$weight between consecutive censuses (the June 1 - Sept 30 window falls entirely within each budget year).
  • Dropped years: the first 15 experimental years after harvest starts, while age structure adjusts.
  • Budget year: May 31 of year \[t\] to May 31 of year \[t+1\], so the 50 experimental years give 49 budget years.

16.4.2 Sources of surplus production

Because every fish is recorded daily, surplus production can be split into its sources:

\[SP_t = G_t + R_t - M_t - L_t\]

  • \[G_t\] (growth): weight gained by stock members, daily weight \[\times\] ggd, summed.
  • \[R_t\] (recruitment): biomass of fish first reaching 200 g peak weight, at the weight they cross.
  • \[M_t\] (natural mortality): biomass of stock fish that die naturally, at their weight at death.
  • \[L_t\] (mass loss): weight lost by stock members other than through death or harvest. The simple \[G + R - M\] identity does not hold without this term because spawning and seasonal mass losses removes weight that ggd does not capture.

The check printed above confirms that this decomposition reproduces \[B_{t+1} - B_t + C_t\] to numerical precision.

16.6 Adult-to-adult stock-recruitment

Recruitment is expressed as the number of adults that survived to spawning size in year Y, defined as spawners(Y) + fish harvested in summer Y−1 (those that would have spawned but were removed). This is paired with the parent spawning stock 3 years earlier (minimum age at maturity), and compared to the 1:1 replacement line to identify surplus production.

# ── Build lagged SR dataset + Ricker fits for one lag value ──────────────────
build_adult_sr <- function(lag) {
  ts <- recruits_adult |>
    mutate(yr_parent = yr - lag) |>
    left_join(
      spawner_log_ts |>
        transmute(hab_type, prop_cold, yr, harvest_rate,
                  n_spawners_parent    = n_spawners,
                  biom_spawners_parent = spawn_biom_log / 1000),
      by = c("hab_type", "prop_cold", "yr_parent" = "yr", "harvest_rate")
    ) |>
    filter(!is.na(n_spawners_parent), n_spawners_parent > 0, n_recruited > 0) %>%
    mutate(hab_type = factor(hab_type, levels = c("Cold only", "Cold + Warm", "Cold + Warm (high P)")))

  summ <- ts |>
    group_by(hab_type, prop_cold, prop_cold_fact, harvest_rate, harvest_rate_fact) |>
    summarise(
      across(c(n_recruited, n_spawners_parent, biom_recruited, biom_spawners_parent),
             list(mean = mean, sd = sd)),
      .groups = "drop"
    ) %>%
    mutate(hab_type = factor(hab_type, levels = c("Cold only", "Cold + Warm", "Cold + Warm (high P)")))

  # Fit Ricker separately per hab_type × prop_cold group
  hab_fits <- ts |>
    group_by(hab_type, prop_cold, prop_cold_fact) |>
    group_split() |>
    purrr::map(\(d) {
      fit_n <- nls(n_recruited ~ a * n_spawners_parent * exp(-b * n_spawners_parent),
                   data = d, start = list(a = 5, b = 0.05))
      fit_b <- nls(biom_recruited ~ a * biom_spawners_parent * exp(-b * biom_spawners_parent),
                   data = d, start = list(a = 5, b = 0.05))
      N_seq <- seq(0, max(d$n_spawners_parent)    , length.out = 300)
      B_seq <- seq(0, max(d$biom_spawners_parent) , length.out = 300)
      list(
        hab_type       = as.character(d$hab_type[1]),
        prop_cold_fact = as.character(d$prop_cold_fact[1]),
        alpha_n = log(coef(fit_n)["a"]), alpha_b = log(coef(fit_b)["a"]),
        beta_n  = coef(fit_n)["b"],      beta_b  = coef(fit_b)["b"],
        s_star_n = log(coef(fit_n)["a"]) / coef(fit_n)["b"],
        s_star_b = log(coef(fit_b)["a"]) / coef(fit_b)["b"],
        pred_n   = tibble(x = N_seq,
                          y = predict(fit_n, list(n_spawners_parent    = N_seq)),
                          hab_type = d$hab_type[1], prop_cold_fact = d$prop_cold_fact[1]),
        pred_b   = tibble(x = B_seq,
                          y = predict(fit_b, list(biom_spawners_parent = B_seq)),
                          hab_type = d$hab_type[1], prop_cold_fact = d$prop_cold_fact[1])
      )
    })

  # Combine predictions, preserving hab_type and prop_cold_fact for faceting
  pred_n <- purrr::map_dfr(hab_fits, \(hf) hf$pred_n) |>
    mutate(hab_type = factor(hab_type, levels = c("Cold only", "Cold + Warm", "Cold + Warm (high P)")))
  pred_b <- purrr::map_dfr(hab_fits, \(hf) hf$pred_b) |>
    mutate(hab_type = factor(hab_type, levels = c("Cold only", "Cold + Warm", "Cold + Warm (high P)")))

  list(
    lag      = lag,
    ts       = ts,
    summ     = summ,
    hab_fits = hab_fits,
    pred_n   = pred_n,
    pred_b   = pred_b
  )
}

16.6.2 Total-adult based

Stock is defined as all adults (>200 g) present at the October 1 census in year Y−lag, with no harvest correction. Recruits are the total adults in the October 1 census of year Y plus fish harvested in summer of year Y (those removed before the census who would otherwise have been counted). Unlike the spawner-based approach, this definition includes adults that did not spawn due to poor condition, capturing condition-driven variation in spawning propensity across habitat scenarios and cold proportions.

get_total_adults <- function(scen) {
  base_cols <- list(
    hab_type          = scen$ht,
    prop_cold         = scen$pc,
    prop_cold_fact    = factor(
      pc_labels_mech[match(scen$pc, pc_levels_mech)],

    ),

    harvest_rate_fact = factor(scen$hr,
                               levels = hr_levels_mech,
                               labels = hr_labels_mech)
  )
  oct_adults <- scen$oct1_adult

  if (is.null(scen$harvest)) {
    oct_adults |>
      right_join(sr_exp_years, by = "yr") |>
      replace_na(list(n_oct1 = 0L, biom_oct1 = 0)) |>
      mutate(n_harv = 0L, biom_harv = 0,
             n_total_adults    = n_oct1,
             biom_total_adults = biom_oct1,

# ── Build lagged SR dataset + Ricker fits ────────────────────────────────────
build_adult_sr_v2 <- function(lag) {
  ts <- total_adults_ts |>
    select(hab_type, prop_cold, prop_cold_fact, harvest_rate, harvest_rate_fact,
           yr, n_recruited = n_total_adults, biom_recruited = biom_total_adults) |>
    mutate(yr_parent = yr - lag) |>
    left_join(
      total_adults_ts |>
        transmute(hab_type, prop_cold, yr, harvest_rate,
                  n_stock    = n_oct1,
                  biom_stock = biom_oct1),
      by = c("hab_type", "prop_cold", "yr_parent" = "yr", "harvest_rate")
    ) |>
    filter(!is.na(n_stock), n_stock > 0, n_recruited > 0) |>
    mutate(hab_type = factor(hab_type, levels = c("Cold only", "Cold + Warm", "Cold + Warm (high P)")))

  summ <- ts |>
    group_by(hab_type, prop_cold, prop_cold_fact, harvest_rate, harvest_rate_fact) |>
    summarise(
      across(c(n_recruited, n_stock, biom_recruited, biom_stock),
             list(mean = mean, sd = sd)),
      .groups = "drop"
    )

  # Fit Ricker separately per hab_type × prop_cold group
  hab_fits <- ts |>
    group_by(hab_type, prop_cold, prop_cold_fact) |>
    group_split() |>
    purrr::map(\(d) {
      fit_n <- tryCatch(
        nls(n_recruited ~ a * n_stock * exp(-b * n_stock),
            data = d, start = list(a = 5, b = 0.05)),
        error = \(e) NULL)
      fit_b <- tryCatch(
        nls(biom_recruited ~ a * biom_stock * exp(-b * biom_stock),
            data = d, start = list(a = 5, b = 0.05)),
        error = \(e) NULL)
      N_seq <- seq(0, max(d$n_stock),    length.out = 300)
      B_seq <- seq(0, max(d$biom_stock), length.out = 300)
      mk_pred <- function(fit, xvar, seq_) {
        if (is.null(fit)) return(
          tibble(x = numeric(), y = numeric(),
                 hab_type = character(), prop_cold_fact = character()))
        tibble(x = seq_,
               y = predict(fit, setNames(list(seq_), xvar)),
               hab_type       = d$hab_type[1],
               prop_cold_fact = d$prop_cold_fact[1])
      }
      list(
        hab_type       = as.character(d$hab_type[1]),
        prop_cold_fact = as.character(d$prop_cold_fact[1]),
        alpha_n  = if (!is.null(fit_n)) log(coef(fit_n)["a"]) else NA,
        alpha_b  = if (!is.null(fit_b)) log(coef(fit_b)["a"]) else NA,
        beta_n   = if (!is.null(fit_n)) coef(fit_n)["b"]      else NA,
        beta_b   = if (!is.null(fit_b)) coef(fit_b)["b"]      else NA,
        s_star_n = if (!is.null(fit_n)) log(coef(fit_n)["a"]) / coef(fit_n)["b"] else NA,
        s_star_b = if (!is.null(fit_b)) log(coef(fit_b)["a"]) / coef(fit_b)["b"] else NA,
        pred_n   = mk_pred(fit_n, "n_stock",    N_seq),
        pred_b   = mk_pred(fit_b, "biom_stock", B_seq)
      )
    })

  # Combine predictions, preserving hab_type and prop_cold_fact for faceting
  pred_n <- purrr::map_dfr(hab_fits, "pred_n") |>
    mutate(hab_type = factor(hab_type, levels = c("Cold only", "Cold + Warm", "Cold + Warm (high P)")))
  pred_b <- purrr::map_dfr(hab_fits, "pred_b") |>
    mutate(hab_type = factor(hab_type, levels = c("Cold only", "Cold + Warm", "Cold + Warm (high P)")))

  list(lag = lag, ts = ts, summ = summ, hab_fits = hab_fits,
       pred_n = pred_n, pred_b = pred_b)
}

## Setup

Most outputs come from the summary objects (`ibm_summary`, `spawn_log`, `harvest_log`). `ibm_long` is used only inside `compute_budget()`, which reduces it to an annual biomass budget for the exploitable stock, so the full individual-level data frame is never stored. ```{r}
#| cache: false

# get experiment start date from base parameters
exp_start_date <- base_params$mindate + years(base_params$nyears_burnin)

# Helper: read one scenario, pull what is needed from the summary objects, and free res.
# ibm_long is never touched, so it is never copied. Extracted objects:
#   oct1_waa   – Oct 1 weight-at-age summary (n, mean, sd) by year x age class,
#                pooled over strategy and patch (ibm_summary is one row per
#                strategy x patch x age_class x month-start)
#   oct1_adult – Oct 1 adults (peak_weight > 200 g): count and biomass (kg) by year
#   age0_nov1  – annual age-0 counts and biomass on Nov 1
#   fry_annual – annual fry birth counts (sum of spawn_log offspring)
#   budget     – annual biomass budget of the exploitable stock (see compute_budget)
#   spawners   – annual spawner counts, size, age, new-entrant fraction and fry (see compute_spawners)
# Requires results generated with the n_adult / biomass_adult_g columns in ibm_summary.

# Annual biomass budget of the exploitable stock (peak_weight >= 200 g, matching
# harvest eligibility), from ibm_long and harvest_log.
#
# Budget year t runs from the census on day-of-year 151 of year t (the last day before
# the harvest window opens on DOY 152, so the census is pre-harvest) to the same census
# in year t + 1. Each fish-day transition d -> d + 1 of a stock member is classified as
#   stay    – survives; weight change is growth
#   die     – natural death on d + 1; weight at death leaves the stock
#   harvest – no ibm_long row on d + 1 and a harvest_log record; harvest weight leaves
# so that, exactly,  B[t+1] - B[t] = G_net + R - M - C  and  SP = B[t+1] - B[t] + C = G_net + R - M.
#   G_gross  – sum of weight x ggd over stock members (growth before spawning losses)
#   L_spawn  – G_gross - G_net: weight lost by stock members other than through death or
#              harvest (spawning); not in the simple SP = G + R - M decomposition
#   R        – biomass of fish at the weight they first reach peak_weight >= 200 g
# peak_weight is reconstructed as the running max of weight within ibm_long, so fish
# already present on the first day lack pre-history; the first years are dropped downstream.
# All biomass in kg.
compute_budget <- function(ibm_long, harvest_log, habitat_df, census_doy = 151L) {
  empty <- tibble(yr = integer(), B = numeric(), C = numeric(), G_net = numeric(),
                  G_gross = numeric(), L_spawn = numeric(), R = numeric(),
                  M = numeric(), B_next = numeric(), SP = numeric())
  if (is.null(ibm_long) || nrow(ibm_long) == 0) return(empty)

  il <- data.table::as.data.table(ibm_long)[, .(pid, dayofsim, weight, ggd, survived)]
  data.table::setorder(il, pid, dayofsim)
  cens <- habitat_df |>
    filter(doy == census_doy, dayofsim >= min(il$dayofsim), dayofsim <= max(il$dayofsim)) |>
    transmute(dayofsim, yr = year(date))
  first_day <- min(il$dayofsim)

  il[, peak := cummax(weight), by = pid]
  il[, `:=`(w_nx   = data.table::shift(weight,   -1L),
            d_nx   = data.table::shift(dayofsim, -1L),
            s_nx   = data.table::shift(survived, -1L),
            pid_nx = data.table::shift(pid,      -1L))]
  il[, nx_ok := !is.na(pid_nx) & pid_nx == pid & d_nx == dayofsim + 1L]
  il[, stock := survived == 1L & peak >= 200]

  # Harvested fish have no row on the harvest day; key their harvest weight to the day before
  hl <- data.table::as.data.table(harvest_log)[, .(pid, dayofsim = dayofsim - 1L, h_w = weight)]
  st <- merge(il[stock == TRUE], hl, by = c("pid", "dayofsim"), all.x = TRUE, sort = FALSE)
  st[, event := data.table::fifelse(nx_ok & s_nx == 1L, "stay",
                data.table::fifelse(nx_ok & s_nx == 0L, "die",
                data.table::fifelse(!is.na(h_w), "harvest", "end")))]
  st <- st[event != "end"]
  st[, w_out := data.table::fifelse(event == "harvest", h_w, w_nx)]
  # Budget year of the flow, indexed by the census that opens it
  st[, t_idx := findInterval(dayofsim, cens$dayofsim)]
  flows <- st[t_idx >= 1L, .(G_net   = sum(w_out - weight),
                             G_gross = sum(weight * ggd),
                             M       = sum(data.table::fifelse(event == "die",     w_out, 0)),
                             C       = sum(data.table::fifelse(event == "harvest", w_out, 0))),
              by = t_idx]

  # Recruits: first alive day at peak >= 200 g (fish present on day 1 are initial stock)
  rec <- unique(il[survived == 1L & peak >= 200 & dayofsim > first_day], by = "pid")
  rec[, t_idx := findInterval(dayofsim - 1L, cens$dayofsim)]
  rec <- rec[t_idx >= 1L, .(R = sum(weight)), by = t_idx]

  bio <- il[dayofsim %in% cens$dayofsim & stock == TRUE, .(B = sum(weight)), by = dayofsim]

  cens |>
    mutate(t_idx = row_number()) |>
    left_join(as_tibble(bio),   by = "dayofsim") |>
    left_join(as_tibble(flows), by = "t_idx") |>
    left_join(as_tibble(rec),   by = "t_idx") |>
    arrange(t_idx) |>
    mutate(across(c(B, G_net, G_gross, M, C, R), \(x) replace_na(x, 0) / 1000),
           B_next  = lead(B),
           L_spawn = G_gross - G_net) |>
    filter(!is.na(B_next)) |>
    mutate(SP = B_next - B + C) |>
    select(yr, B, C, G_net, G_gross, L_spawn, R, M, B_next, SP)
}

# Annual spawner summary from spawn_log, one row per spawning year in the experimental period.
# A spawner is a "new entrant" if it first reached peak_weight >= 200 g (reconstructed as the
# running max of weight in ibm_long) after the most recent harvest-window opening (DOY 152),
# i.e. it was never eligible for harvest before this spawning season. Fish already >= 200 g on
# the first day of ibm_long have unknown crossing dates and count as not new.
#   n_spawners, spawn_biom_kg – distinct parents and their summed weight at spawning
#   mean_wt, mean_age         – per spawning event; age from fish_registry birth dates
#   frac_new                  – fraction of distinct spawners that were new entrants
#   fry                       – sum of n_offspring
compute_spawners <- function(ibm_long, spawn_log, fish_registry, habitat_df, first_date,
                             window_open_doy = 152L) {
  empty <- tibble(yr = integer(), n_spawners = integer(), n_events = integer(),
                  mean_wt = numeric(), mean_age = numeric(), frac_new = numeric(),
                  spawn_biom_kg = numeric(), fry = numeric())
  if (is.null(ibm_long) || nrow(ibm_long) == 0 || nrow(spawn_log) == 0) return(empty)

  il <- data.table::as.data.table(ibm_long)[, .(pid, dayofsim, weight)]
  data.table::setorder(il, pid, dayofsim)
  il[, peak := cummax(weight), by = pid]
  first_day <- min(il$dayofsim)
  cross <- il[peak >= 200, .(cross_day = min(dayofsim)), by = pid]
  cross[cross_day == first_day, cross_day := NA_integer_]
  rm(il)

  win_open <- habitat_df$dayofsim[habitat_df$doy == window_open_doy]
  spawn_log |>
    mutate(date = habitat_df$date[dayofsim], yr = year(date)) |>
    filter(date >= first_date) |>
    left_join(select(fish_registry, pid, birth_dayofsim), by = c("parent_pid" = "pid")) |>
    left_join(as_tibble(cross), by = c("parent_pid" = "pid")) |>
    mutate(prev_open   = win_open[findInterval(dayofsim, win_open)],
           age         = (dayofsim - birth_dayofsim) / 365.25,
           new_entrant = !is.na(cross_day) & cross_day >= prev_open) |>
    group_by(yr) |>
    summarise(n_spawners    = n_distinct(parent_pid),
              n_events      = n(),
              mean_wt       = mean(weight),
              mean_age      = mean(age),
              frac_new      = n_distinct(parent_pid[new_entrant]) / n_distinct(parent_pid),
              spawn_biom_kg = sum(weight[!duplicated(parent_pid)]) / 1000,
              fry           = sum(n_offspring),
              .groups = "drop")
}

load_scen <- function(path, ht, pc, ac, aw, hr, pcmax_cold, pcmax_warm) {
  res <- readRDS(path)
  stopifnot("ibm_summary lacks patch/age_class/n_adult; re-run simulation" =
              all(c("patch", "age_class", "n_adult") %in% names(res$ibm_summary)))

  sm <- res$ibm_summary |>
    filter(date >= exp_start_date)

  sm_oct1 <- sm |> filter(month(date) == 10, day(date) == 1)

  # Pool strategy x patch groups within each age class. Group means are
  # n-weighted (sum of biomass / sum of n); pooled variance combines within-group
  # and between-group sums of squares. sd_weight is NA for single-fish groups;
  # their (n - 1) weight is 0 anyway.
  oct1_waa <- sm_oct1 |>
    filter(n_alive > 0) |>
    mutate(ss_within = (n_alive - 1) * replace_na(sd_weight, 0)^2,
           ss_mean   = n_alive * mean_weight^2) |>
    group_by(yr = year(date), age_class) |>
    summarise(
      n           = sum(n_alive),
      mean_weight = sum(biomass_g) / n,
      sd_weight   = sqrt(pmax(sum(ss_within) + sum(ss_mean) - n * mean_weight^2, 0) /
                           pmax(n - 1, 1)),
      .groups = "drop"
    )

  # Sum across strategy, patch, and age class
  oct1_adult <- sm_oct1 |>
    group_by(yr = year(date)) |>
    summarise(n_oct1    = sum(n_adult),
              biom_oct1 = sum(biomass_adult_g) / 1000,  # g -> kg
              .groups = "drop")

  # Age-0 only; summed across strategy and patch
  age0_nov1 <- sm |>
    filter(month(date) == 11, day(date) == 1, age_class == 0) |>
    group_by(yr = year(date)) |>
    summarise(n_age0 = sum(n_alive), biom_age0 = sum(biomass_g), .groups = "drop")

  fry_annual <- res$spawn_log |>
    mutate(date = res$habitat_df$date[dayofsim]) |>
    filter(date >= exp_start_date) |>
    group_by(yr = year(date)) |>
    summarise(n_fry = sum(n_offspring), .groups = "drop")

  list(oct1_waa = oct1_waa, oct1_adult = oct1_adult,
       age0_nov1 = age0_nov1, fry_annual = fry_annual,
       budget = compute_budget(res$ibm_long, res$harvest_log, res$habitat_df),
       spawners = compute_spawners(res$ibm_long, res$spawn_log, res$fish_registry,
                                   res$habitat_df, exp_start_date),
       harvest = res$harvest_log, spawn_log = res$spawn_log,
       ht = ht, pc = pc, ac = ac, aw = aw, hr = hr,
       pcmax_cold = pcmax_cold, pcmax_warm = pcmax_warm)
}

## Surplus production: biomass budget

Surplus production is the biomass the stock adds in a year beyond what is needed to replace itself, i.e., the yield that could be taken while leaving the stock the same size. Because the IBM tracks every fish, it can be measured directly with a biomass budget, with no stock-recruitment model and no choice of lag:

$$SP_t = B_{t+1} - B_t + C_t$$

where $$B_t$$ is the biomass of the exploitable stock at a fixed census date and $$C_t$$ is the biomass harvested between censuses. At equilibrium ($$B_{t+1} = B_t$$) surplus production equals catch by definition, which is why the Ricker-based estimates further below track the harvest summaries so closely.

Implementation (see `compute_budget()` in Setup):

- **Stock**: fish with $$peak\_weight \geq 200$$ g, matching harvest eligibility. `ibm_long` has no `peak_weight` column, so it is reconstructed as the running maximum of each fish's weight within `ibm_long`.
- **Census**: day of year 151, the last day before the harvest window opens (DOY 152), every year. `ibm_long` is recorded after that day's harvest, so a census on DOY 152 would already be missing the first day's catch; DOY 151 keeps the budget closed.
- **Catch**: sum of `harvest_log$weight` between consecutive censuses (the June 1 - Sept 30 window falls entirely within each budget year).
- **Dropped years**: the first 15 experimental years after harvest starts, while age structure adjusts.
- **Budget year**: May 31 of year $$t$$ to May 31 of year $$t+1$$, so the 50 experimental years give 49 budget years.

### Sources of surplus production

Because every fish is recorded daily, surplus production can be split into its sources:

$$SP_t = G_t + R_t - M_t - L_t$$

- $$G_t$$ (growth): weight gained by stock members, daily weight $$\times$$ `ggd`, summed.
- $$R_t$$ (recruitment): biomass of fish first reaching 200 g peak weight, at the weight they cross.
- $$M_t$$ (natural mortality): biomass of stock fish that die naturally, at their weight at death.
- $$L_t$$ (mass loss): weight lost by stock members other than through death or harvest. The simple $$G + R - M$$ identity does not hold without this term because spawning and seasonal mass losses removes weight that `ggd` does not capture.

The check printed above confirms that this decomposition reproduces $$B_{t+1} - B_t + C_t$$ to numerical precision.

## Adult-to-adult stock-recruitment

Recruitment is expressed as the number of adults that survived to spawning size in year Y, defined as spawners(Y) + fish harvested in summer Y−1 (those that would have spawned but were removed). This is paired with the parent spawning stock 3 years earlier (minimum age at maturity), and compared to the 1:1 replacement line to identify surplus production.

# ── Build lagged SR dataset + Ricker fits for one lag value ──────────────────
build_adult_sr <- function(lag) {
  ts <- recruits_adult |>
    mutate(yr_parent = yr - lag) |>
    left_join(
      spawner_log_ts |>
        transmute(hab_type, prop_cold, yr, harvest_rate,
                  n_spawners_parent    = n_spawners,
                  biom_spawners_parent = spawn_biom_log / 1000),
      by = c("hab_type", "prop_cold", "yr_parent" = "yr", "harvest_rate")
    ) |>
    filter(!is.na(n_spawners_parent), n_spawners_parent > 0, n_recruited > 0) %>%
    mutate(hab_type = factor(hab_type, levels = c("Cold only", "Cold + Warm", "Cold + Warm (high P)")))

  summ <- ts |>
    group_by(hab_type, prop_cold, prop_cold_fact, harvest_rate, harvest_rate_fact) |>
    summarise(
      across(c(n_recruited, n_spawners_parent, biom_recruited, biom_spawners_parent),
             list(mean = mean, sd = sd)),
      .groups = "drop"
    ) %>%
    mutate(hab_type = factor(hab_type, levels = c("Cold only", "Cold + Warm", "Cold + Warm (high P)")))

  # Fit Ricker separately per hab_type × prop_cold group
  hab_fits <- ts |>
    group_by(hab_type, prop_cold, prop_cold_fact) |>
    group_split() |>
    purrr::map(\(d) {
      fit_n <- nls(n_recruited ~ a * n_spawners_parent * exp(-b * n_spawners_parent),
                   data = d, start = list(a = 5, b = 0.05))
      fit_b <- nls(biom_recruited ~ a * biom_spawners_parent * exp(-b * biom_spawners_parent),
                   data = d, start = list(a = 5, b = 0.05))
      N_seq <- seq(0, max(d$n_spawners_parent)    , length.out = 300)
      B_seq <- seq(0, max(d$biom_spawners_parent) , length.out = 300)
      list(
        hab_type       = as.character(d$hab_type[1]),
        prop_cold_fact = as.character(d$prop_cold_fact[1]),
        alpha_n = log(coef(fit_n)["a"]), alpha_b = log(coef(fit_b)["a"]),
        beta_n  = coef(fit_n)["b"],      beta_b  = coef(fit_b)["b"],
        s_star_n = log(coef(fit_n)["a"]) / coef(fit_n)["b"],
        s_star_b = log(coef(fit_b)["a"]) / coef(fit_b)["b"],
        pred_n   = tibble(x = N_seq,
                          y = predict(fit_n, list(n_spawners_parent    = N_seq)),
                          hab_type = d$hab_type[1], prop_cold_fact = d$prop_cold_fact[1]),
        pred_b   = tibble(x = B_seq,
                          y = predict(fit_b, list(biom_spawners_parent = B_seq)),
                          hab_type = d$hab_type[1], prop_cold_fact = d$prop_cold_fact[1])
      )
    })

  # Combine predictions, preserving hab_type and prop_cold_fact for faceting
  pred_n <- purrr::map_dfr(hab_fits, \(hf) hf$pred_n) |>
    mutate(hab_type = factor(hab_type, levels = c("Cold only", "Cold + Warm", "Cold + Warm (high P)")))
  pred_b <- purrr::map_dfr(hab_fits, \(hf) hf$pred_b) |>
    mutate(hab_type = factor(hab_type, levels = c("Cold only", "Cold + Warm", "Cold + Warm (high P)")))

  list(
    lag      = lag,
    ts       = ts,
    summ     = summ,
    hab_fits = hab_fits,
    pred_n   = pred_n,
    pred_b   = pred_b
  )
}

### Total-adult based

Stock is defined as all adults (>200 g) present at the October 1 census in year Y−lag, with no harvest correction. Recruits are the total adults in the October 1 census of year Y plus fish harvested in summer of year Y (those removed before the census who would otherwise have been counted). Unlike the spawner-based approach, this definition includes adults that did not spawn due to poor condition, capturing condition-driven variation in spawning propensity across habitat scenarios and cold proportions.

#### Derive stock-recruitment curves

Derive adult-to-adult stock-recruitment curves over 4 different lag periods (2–5 years), for abundance and biomass separately.

# ── Total adult abundance and biomass per scenario per year ───────────────────
# Stock  (X): Oct 1 census adults > 200 g only — no harvest correction
# Recruits (Y): Oct 1 census adults + same-year harvest (removed before census)
get_total_adults <- function(scen) {
  base_cols <- list(
    hab_type          = scen$ht,
    prop_cold         = scen$pc,
    prop_cold_fact    = factor(
      pc_labels_mech[match(scen$pc, pc_levels_mech)],
      levels = pc_labels_mech
    ),
    harvest_rate      = scen$hr,
    harvest_rate_fact = factor(scen$hr,
                               levels = hr_levels_mech,
                               labels = hr_labels_mech)
  )
  oct_adults <- scen$oct1_adult

  if (is.null(scen$harvest)) {
    oct_adults |>
      right_join(sr_exp_years, by = "yr") |>
      replace_na(list(n_oct1 = 0L, biom_oct1 = 0)) |>
      mutate(n_harv = 0L, biom_harv = 0,
             n_total_adults    = n_oct1,
             biom_total_adults = biom_oct1,
             !!!base_cols,
             hab_type = factor(hab_type, levels = c("Cold only", "Cold + Warm", "Cold + Warm (high P)")))
  } else {
    harv_yr <- scen$harvest |>
      group_by(yr = year) |>
      summarise(n_harv    = n(),
                biom_harv = sum(weight) / 1000,
                .groups   = "drop")
    oct_adults |>
      right_join(sr_exp_years, by = "yr") |>
      left_join(harv_yr, by = "yr") |>
      replace_na(list(n_oct1 = 0L, biom_oct1 = 0, n_harv = 0L, biom_harv = 0)) |>
      mutate(n_total_adults    = n_oct1    + n_harv,
             biom_total_adults = biom_oct1 + biom_harv,
             !!!base_cols) |>
      mutate(hab_type = factor(hab_type, levels = c("Cold only", "Cold + Warm", "Cold + Warm (high P)")))
  }
}

total_adults_ts <- purrr::map_dfr(ibm_list_mech, get_total_adults)

# ── Build lagged SR dataset + Ricker fits ────────────────────────────────────
build_adult_sr_v2 <- function(lag) {
  ts <- total_adults_ts |>
    select(hab_type, prop_cold, prop_cold_fact, harvest_rate, harvest_rate_fact,
           yr, n_recruited = n_total_adults, biom_recruited = biom_total_adults) |>
    mutate(yr_parent = yr - lag) |>
    left_join(
      total_adults_ts |>
        transmute(hab_type, prop_cold, yr, harvest_rate,
                  n_stock    = n_oct1,
                  biom_stock = biom_oct1),
      by = c("hab_type", "prop_cold", "yr_parent" = "yr", "harvest_rate")
    ) |>
    filter(!is.na(n_stock), n_stock > 0, n_recruited > 0) |>
    mutate(hab_type = factor(hab_type, levels = c("Cold only", "Cold + Warm", "Cold + Warm (high P)")))

  summ <- ts |>
    group_by(hab_type, prop_cold, prop_cold_fact, harvest_rate, harvest_rate_fact) |>
    summarise(
      across(c(n_recruited, n_stock, biom_recruited, biom_stock),
             list(mean = mean, sd = sd)),
      .groups = "drop"
    )

  # Fit Ricker separately per hab_type × prop_cold group
  hab_fits <- ts |>
    group_by(hab_type, prop_cold, prop_cold_fact) |>
    group_split() |>
    purrr::map(\(d) {
      fit_n <- tryCatch(
        nls(n_recruited ~ a * n_stock * exp(-b * n_stock),
            data = d, start = list(a = 5, b = 0.05)),
        error = \(e) NULL)
      fit_b <- tryCatch(
        nls(biom_recruited ~ a * biom_stock * exp(-b * biom_stock),
            data = d, start = list(a = 5, b = 0.05)),
        error = \(e) NULL)
      N_seq <- seq(0, max(d$n_stock),    length.out = 300)
      B_seq <- seq(0, max(d$biom_stock), length.out = 300)
      mk_pred <- function(fit, xvar, seq_) {
        if (is.null(fit)) return(
          tibble(x = numeric(), y = numeric(),
                 hab_type = character(), prop_cold_fact = character()))
        tibble(x = seq_,
               y = predict(fit, setNames(list(seq_), xvar)),
               hab_type       = d$hab_type[1],
               prop_cold_fact = d$prop_cold_fact[1])
      }
      list(
        hab_type       = as.character(d$hab_type[1]),
        prop_cold_fact = as.character(d$prop_cold_fact[1]),
        alpha_n  = if (!is.null(fit_n)) log(coef(fit_n)["a"]) else NA,
        alpha_b  = if (!is.null(fit_b)) log(coef(fit_b)["a"]) else NA,
        beta_n   = if (!is.null(fit_n)) coef(fit_n)["b"]      else NA,
        beta_b   = if (!is.null(fit_b)) coef(fit_b)["b"]      else NA,
        s_star_n = if (!is.null(fit_n)) log(coef(fit_n)["a"]) / coef(fit_n)["b"] else NA,
        s_star_b = if (!is.null(fit_b)) log(coef(fit_b)["a"]) / coef(fit_b)["b"] else NA,
        pred_n   = mk_pred(fit_n, "n_stock",    N_seq),
        pred_b   = mk_pred(fit_b, "biom_stock", B_seq)
      )
    })

  # Combine predictions, preserving hab_type and prop_cold_fact for faceting
  pred_n <- purrr::map_dfr(hab_fits, "pred_n") |>
    mutate(hab_type = factor(hab_type, levels = c("Cold only", "Cold + Warm", "Cold + Warm (high P)")))
  pred_b <- purrr::map_dfr(hab_fits, "pred_b") |>
    mutate(hab_type = factor(hab_type, levels = c("Cold only", "Cold + Warm", "Cold + Warm (high P)")))

  list(lag = lag, ts = ts, summ = summ, hab_fits = hab_fits,
       pred_n = pred_n, pred_b = pred_b)
}

#### Surplus production vs. harvest rate

Surplus production per year is the signed distance above the 1:1 line: total recruited adults (Oct 1 census + same-year harvest) minus the total adult stock in year Y−lag. Positive = net gain; zero = replacement; negative = decline.
