13.1 Setup

Load relevant .rds objects and bind directly into a list. Using a helper function ensures each res object is freed from memory as soon as ibm_long is extracted, avoiding duplicate references that would force R to copy the data frames.

# get experiment start date from base parameters
exp_start_date <- base_params$mindate + years(base_params$nyears_burnin)

# Helper: read one scenario, extract ibm_long summaries, and free res before returning.
# Rather than storing the full ibm_long data frame (very large), we pre-extract the
# three filtered/aggregated subsets that are actually used downstream:
#   oct1_exp   – Oct 1 weight-at-age rows for the experimental period
#   age0_nov1  – annual age-0 counts and biomass on Nov 1 (used in both the
#                Recruitment and Stock-Recruitment sections)
#   fry_annual – annual fry birth counts for the experimental period
load_scen <- function(path, ht, pc, ac, aw, hr, pcmax_cold, pcmax_warm) {
  res <- readRDS(path)
  ibm <- res$ibm_long

  oct1_exp <- ibm |>
    filter(date >= exp_start_date,
           month(date) == 10, day(date) == 1,
           survived == 1) |>
    mutate(yr = year(date), age_class = floor(age)) |>
    select(yr, age_class, weight)

  age0_nov1 <- ibm |>
    filter(date >= exp_start_date,
           month(date) == 11, day(date) == 1,
           survived == 1, floor(age) == 0) |>
    group_by(yr = year(date)) |>
    summarise(n_age0 = n(), biom_age0 = sum(weight), .groups = "drop")

  fry_annual <- ibm |>
    filter(date >= exp_start_date,
           !is.na(parent_pid),
           dayofsim == birth_dayofsim) |>
    count(yr = year(date), name = "n_fry")

  list(oct1_exp = oct1_exp, age0_nov1 = age0_nov1, fry_annual = fry_annual,
       harvest = res$harvest_log, spawn_log = res$spawn_log,
       ht = ht, pc = pc, ac = ac, aw = aw, hr = hr,
       pcmax_cold = pcmax_cold, pcmax_warm = pcmax_warm)
}

13.3 Recruitment

The number of fry produced is a function of adult size and abundance (fecundity), which are driven by warm and cold habitat availability. Importantly, when only cold habitat is available, the number of fry produced declines ~linearly with cold habitat availability. # ── Annual age-0 recruitment time series ─────────────────────────────────────
# Zero-fill missing years so post-crash scenarios contribute zeros for the
# remainder of the experimental period rather than being silently excluded.
all_exp_years <- tibble(
  yr = seq(year(exp_start_date),
           year(exp_start_date) + base_params$nyears_experiment - 1L)
)

get_annual_age0 <- function(scen) {
  scen$age0_nov1 |>
    select(yr, n_age0) |>
    right_join(all_exp_years, by = "yr") |>
    replace_na(list(n_age0 = 0L)) |>
    mutate(
      hab_type          = scen$ht,
      prop_cold         = scen$pc,
      harvest_rate      = scen$hr,
      harvest_rate_fact = factor(scen$hr,
                                 levels = hr_levels_mech,
                                 labels = hr_labels_mech),

      prop_cold_fact    = factor(
        pc_labels_mech[match(scen$pc, pc_levels_mech)],
        levels = pc_labels_mech
      )
    )
}

# ── Build lagged SR dataset + Ricker fits for one lag value ──────────────────
build_adult_sr <- function(lag) {
  ts <- recruits_adult |>
    mutate(yr_parent = yr - lag) |>
    left_join(
      spawner_log_ts |>
        transmute(hab_type, prop_cold, yr, harvest_rate,
                  n_spawners_parent    = n_spawners,
                  biom_spawners_parent = spawn_biom_log / 1000),
      by = c("hab_type", "prop_cold", "yr_parent" = "yr", "harvest_rate")
    ) |>
    filter(!is.na(n_spawners_parent), n_spawners_parent > 0, n_recruited > 0) %>%
    mutate(hab_type = factor(hab_type, levels = c("Cold only", "Cold + Warm")))

  summ <- ts |>
    group_by(hab_type, prop_cold, prop_cold_fact, harvest_rate, harvest_rate_fact) |>
    summarise(
      across(c(n_recruited, n_spawners_parent, biom_recruited, biom_spawners_parent),
             list(mean = mean, sd = sd)),
      .groups = "drop"
    ) %>%
    mutate(hab_type = factor(hab_type, levels = c("Cold only", "Cold + Warm")))

  # Fit Ricker separately per hab_type × prop_cold group
  hab_fits <- ts |>
    group_by(hab_type, prop_cold, prop_cold_fact) |>
    group_split() |>
    purrr::map(\(d) {
      fit_n <- nls(n_recruited ~ a * n_spawners_parent * exp(-b * n_spawners_parent),
                   data = d, start = list(a = 5, b = 0.05))
      fit_b <- nls(biom_recruited ~ a * biom_spawners_parent * exp(-b * biom_spawners_parent),
                   data = d, start = list(a = 5, b = 0.05))
      N_seq <- seq(0, max(d$n_spawners_parent)    , length.out = 300)
      B_seq <- seq(0, max(d$biom_spawners_parent) , length.out = 300)
      list(
        hab_type       = as.character(d$hab_type[1]),
        prop_cold_fact = as.character(d$prop_cold_fact[1]),
        alpha_n = log(coef(fit_n)["a"]), alpha_b = log(coef(fit_b)["a"]),
        beta_n  = coef(fit_n)["b"],      beta_b  = coef(fit_b)["b"],
        s_star_n = log(coef(fit_n)["a"]) / coef(fit_n)["b"],
        s_star_b = log(coef(fit_b)["a"]) / coef(fit_b)["b"],
        pred_n   = tibble(x = N_seq,
                          y = predict(fit_n, list(n_spawners_parent    = N_seq)),
                          hab_type = d$hab_type[1], prop_cold_fact = d$prop_cold_fact[1]),
        pred_b   = tibble(x = B_seq,
                          y = predict(fit_b, list(biom_spawners_parent = B_seq)),
                          hab_type = d$hab_type[1], prop_cold_fact = d$prop_cold_fact[1])
      )
    })

  # Combine predictions, preserving hab_type and prop_cold_fact for faceting
  pred_n <- purrr::map_dfr(hab_fits, \(hf) hf$pred_n) |>
    mutate(hab_type = factor(hab_type, levels = c("Cold only", "Cold + Warm")))
  pred_b <- purrr::map_dfr(hab_fits, \(hf) hf$pred_b) |>
    mutate(hab_type = factor(hab_type, levels = c("Cold only", "Cold + Warm")))

  list(
    lag      = lag,
    ts       = ts,
    summ     = summ,
    hab_fits = hab_fits,
    pred_n   = pred_n,
    pred_b   = pred_b
  )
}

# Helper: read one scenario, extract ibm_long summaries, and free res before returning.
# Rather than storing the full ibm_long data frame (very large), we pre-extract the
# three filtered/aggregated subsets that are actually used downstream:
#   oct1_exp   – Oct 1 weight-at-age rows for the experimental period
#   age0_nov1  – annual age-0 counts and biomass on Nov 1 (used in both the
#                Recruitment and Stock-Recruitment sections)
#   fry_annual – annual fry birth counts for the experimental period
load_scen <- function(path, ht, pc, ac, aw, hr, pcmax_cold, pcmax_warm) {
  res <- readRDS(path)
  ibm <- res$ibm_long

  oct1_exp <- ibm |>
    filter(date >= exp_start_date,
           month(date) == 10, day(date) == 1,
           survived == 1) |>
    mutate(yr = year(date), age_class = floor(age)) |>
    select(yr, age_class, weight)

  age0_nov1 <- ibm |>
    filter(date >= exp_start_date,
           month(date) == 11, day(date) == 1,
           survived == 1, floor(age) == 0) |>
    group_by(yr = year(date)) |>
    summarise(n_age0 = n(), biom_age0 = sum(weight), .groups = "drop")

  fry_annual <- ibm |>
    filter(date >= exp_start_date,
           !is.na(parent_pid),
           dayofsim == birth_dayofsim) |>
    count(yr = year(date), name = "n_fry")

  list(oct1_exp = oct1_exp, age0_nov1 = age0_nov1, fry_annual = fry_annual,
       harvest = res$harvest_log, spawn_log = res$spawn_log,
       ht = ht, pc = pc, ac = ac, aw = aw, hr = hr,
       pcmax_cold = pcmax_cold, pcmax_warm = pcmax_warm)
}

## Adult-to-juvenile stock-recruitment

Two measures of spawner biomass are compared: (1) the exact biomass recorded in the spawn log (sum of unique spawner weights per year), and (2) an estimate from the IBM state on April 15 — the total weight of all surviving fish heavier than 200 g just before the spawning window opens. Recruitment is age-0 abundance and biomass on November 1.

# Each parent may appear more than once per year (multiple spawning attempts);
# average their weight within year before summing to avoid double-counting mass.
get_annual_spawner_log <- function(scen) {
  scen$spawn_log |>
    mutate(date = base_params$mindate + (dayofsim - 1L),
           yr   = year(date)) |>
    filter(date >= exp_start_date) |>
    group_by(yr, parent_pid) |>
    summarise(parent_wt = mean(weight), .groups = "drop") |>
    group_by(yr) |>
    summarise(spawn_biom_log = sum(parent_wt),
              n_spawners     = n_distinct(parent_pid),
              .groups = "drop") |>
    right_join(sr_exp_years, by = "yr") |>
    replace_na(list(spawn_biom_log = 0, n_spawners = 0L)) |>

# x_kg / y_kg: set TRUE to divide that axis by 1000 (g -> kg); FALSE for counts.
sr_panel <- function(x_raw, y_raw, x_summ, y_summ,
                     x_sd_summ, y_sd_summ, x_lab, y_lab, title,
                     x_kg = FALSE, y_kg = FALSE) {

  y_fn <- if (y_kg) kg else identity

Recruitment is expressed as the number of adults that survived to spawning size in year Y, defined as spawners(Y) + fish harvested in summer Y−1 (those that would have spawned but were removed). This is paired with the parent spawning stock 3 years earlier (minimum age at maturity), and compared to the 1:1 replacement line to identify surplus production.


### Derive stock-recruitment curves

Derive adult-to-adult stock-recruitment curves over 4 different lag periods (2-5 years), for abundance and biomass separately.

get_annual_harvest_nb <- function(scen) {
  base_cols <- list(
    hab_type       = scen$ht,
    prop_cold      = scen$pc,
    prop_cold_fact = factor(
      pc_labels_mech[match(scen$pc, pc_levels_mech)],
      levels = pc_labels_mech
    ),
    harvest_rate      = scen$hr,
    harvest_rate_fact = factor(scen$hr,
                               levels = hr_levels_mech,
                               labels = hr_labels_mech)
  )
  if (is.null(scen$harvest)) {
    sr_exp_years |>
      mutate(n_harvested = 0L, biom_harvested = 0,
             !!!base_cols)
  } else {
    scen$harvest |>
      group_by(yr = year) |>
      summarise(n_harvested    = n(),
                biom_harvested = sum(weight) / 1000,   # g -> kg
                .groups = "drop") |>
      right_join(sr_exp_years, by = "yr") |>
      replace_na(list(n_harvested = 0L, biom_harvested = 0)) |>
      mutate(!!!base_cols) |>
      mutate(hab_type = factor(hab_type, levels = c("Cold only", "Cold + Warm")))
  }
}

# ── Recruited adults in year Y = spawners(Y) + harvest(summer Y−1) ────────────
recruits_adult <- spawner_log_ts |>
  select(hab_type, prop_cold, prop_cold_fact, yr, harvest_rate, harvest_rate_fact,
         n_spawners, biom_spawners = spawn_biom_log) |>

# ── Build lagged SR dataset + Ricker fits for one lag value ──────────────────
build_adult_sr <- function(lag) {
  ts <- recruits_adult |>
    mutate(yr_parent = yr - lag) |>
    left_join(
      spawner_log_ts |>
        transmute(hab_type, prop_cold, yr, harvest_rate,
                  n_spawners_parent    = n_spawners,
                  biom_spawners_parent = spawn_biom_log / 1000),
      by = c("hab_type", "prop_cold", "yr_parent" = "yr", "harvest_rate")
    ) |>
    filter(!is.na(n_spawners_parent), n_spawners_parent > 0, n_recruited > 0) %>%
    mutate(hab_type = factor(hab_type, levels = c("Cold only", "Cold + Warm")))

  summ <- ts |>
    group_by(hab_type, prop_cold, prop_cold_fact, harvest_rate, harvest_rate_fact) |>
    summarise(
      across(c(n_recruited, n_spawners_parent, biom_recruited, biom_spawners_parent),
             list(mean = mean, sd = sd)),
      .groups = "drop"
    ) %>%
    mutate(hab_type = factor(hab_type, levels = c("Cold only", "Cold + Warm")))

  # Fit Ricker separately per hab_type × prop_cold group
  hab_fits <- ts |>
    group_by(hab_type, prop_cold, prop_cold_fact) |>
    group_split() |>
    purrr::map(\(d) {
      fit_n <- nls(n_recruited ~ a * n_spawners_parent * exp(-b * n_spawners_parent),
                   data = d, start = list(a = 5, b = 0.05))
      fit_b <- nls(biom_recruited ~ a * biom_spawners_parent * exp(-b * biom_spawners_parent),
                   data = d, start = list(a = 5, b = 0.05))
      N_seq <- seq(0, max(d$n_spawners_parent)    , length.out = 300)
      B_seq <- seq(0, max(d$biom_spawners_parent) , length.out = 300)
      list(
        hab_type       = as.character(d$hab_type[1]),
        prop_cold_fact = as.character(d$prop_cold_fact[1]),
        alpha_n = log(coef(fit_n)["a"]), alpha_b = log(coef(fit_b)["a"]),
        beta_n  = coef(fit_n)["b"],      beta_b  = coef(fit_b)["b"],
        s_star_n = log(coef(fit_n)["a"]) / coef(fit_n)["b"],
        s_star_b = log(coef(fit_b)["a"]) / coef(fit_b)["b"],
        pred_n   = tibble(x = N_seq,
                          y = predict(fit_n, list(n_spawners_parent    = N_seq)),
                          hab_type = d$hab_type[1], prop_cold_fact = d$prop_cold_fact[1]),
        pred_b   = tibble(x = B_seq,
                          y = predict(fit_b, list(biom_spawners_parent = B_seq)),
                          hab_type = d$hab_type[1], prop_cold_fact = d$prop_cold_fact[1])
      )
    })

  # Combine predictions, preserving hab_type and prop_cold_fact for faceting
  pred_n <- purrr::map_dfr(hab_fits, \(hf) hf$pred_n) |>
    mutate(hab_type = factor(hab_type, levels = c("Cold only", "Cold + Warm")))
  pred_b <- purrr::map_dfr(hab_fits, \(hf) hf$pred_b) |>
    mutate(hab_type = factor(hab_type, levels = c("Cold only", "Cold + Warm")))

  list(
    lag      = lag,
    ts       = ts,
    summ     = summ,
    hab_fits = hab_fits,
    pred_n   = pred_n,
    pred_b   = pred_b
  )
}

sr_by_lag <- purrr::map(2:5, build_adult_sr)
