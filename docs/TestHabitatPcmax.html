# Helper: read one scenario, extract ibm_long and spawn_log
load_scen <- function(path, pc, ht, ac, aw, pcmax_cold, pcmax_warm) {
  res <- readRDS(path)
  list(
    df         = res$ibm_long,
    spawn_log  = res$spawn_log,
    pc         = pc,
    ht         = ht,
    ac         = ac,
    aw         = aw,
    pcmax_cold = pcmax_cold,
    pcmax_warm = pcmax_warm
  )
}

ibm_list <- list(
  load_scen("results/Sensitivity_ColdOnly_Pfixedp_05p.rds", 0.5, "Cold only",   1.0, 0.0, 0.5, NA_real_),
  load_scen("results/Sensitivity_ColdOnly_Pfixedp_06p.rds", 0.6, "Cold only",   1.0, 0.0, 0.6, NA_real_),
  load_scen("results/Sensitivity_ColdOnly_Pfixedp_07p.rds", 0.7, "Cold only",   1.0, 0.0, 0.7, NA_real_),
  load_scen("results/Sensitivity_ColdWarm_Pfixedp_05p.rds", 0.5, "Cold + Warm", 1.0, 1.0, 0.5, 0.5),
  load_scen("results/Sensitivity_ColdWarm_Pfixedp_06p.rds", 0.6, "Cold + Warm", 1.0, 1.0, 0.6, 0.6),
  load_scen("results/Sensitivity_ColdWarm_Pfixedp_07p.rds", 0.7, "Cold + Warm", 1.0, 1.0, 0.7, 0.7)

pcmax_levels <- c("0.5", "0.6", "0.7")
hab_levels   <- c("Cold only", "Cold + Warm")

hab_fills    <- c("Cold only" = "#6BAED6", "Cold + Warm" = "#FC8D59")
pcmax_cols   <- c("0.5" = "#74C476", "0.6" = "#238B45", "0.7" = "#00441B")
pcmax_ltys   <- c("0.5" = "solid",  "0.6" = "dashed", "0.7" = "dotted")

# Oct 1 census: experimental period, survivors only
exp_oct1 <- purrr::map_dfr(ibm_list, function(s) {
  s$df |>
    filter(date >= exp_start_date,
           month(date) == 10, day(date) == 1,
           survived == 1) |>
    mutate(
      pcmax_val = factor(s$pc, levels = pcmax_levels),
      hab_type  = s$ht,
      year      = year(date),
      age_class = floor(age)
    )
}) |>
  mutate(hab_type = factor(hab_type, levels = hab_levels))

pop_yr <- exp_oct1 |>
  group_by(pcmax_val, hab_type, year) |>
  summarise(n = n(), biomass_kg = sum(weight) / 1000, .groups = "drop")

pop_sum <- pop_yr |>
  group_by(pcmax_val, hab_type) |>
  summarise(mean_n = mean(n), sd_n = sd(n),
            mean_b = mean(biomass_kg), sd_b = sd(biomass_kg), .groups = "drop")

age0_sum <- exp_oct1 |>
  filter(age_class == 0) |>
  group_by(pcmax_val, hab_type, year) |>
  summarise(n_age0 = n(), .groups = "drop") |>
  group_by(pcmax_val, hab_type) |>
  summarise(mean_n0 = mean(n_age0), sd_n0 = sd(n_age0), .groups = "drop")

waa <- exp_oct1 |>
  filter(age_class >= 0, age_class <= 15) |>
  group_by(pcmax_val, hab_type, age_class) |>
  summarise(mean_wt = mean(weight), sd_wt = sd(weight), n = n(), .groups = "drop") |>
  filter(n >= 10)

spawn_yr <- purrr::map_dfr(ibm_list, function(s) {
  s$spawn_log |>
    mutate(date = sim_params$mindate + (dayofsim - 1L), yr = year(date)) |>
    filter(date >= exp_start_date) |>
    group_by(yr, parent_pid) |>
    summarize(parent_wt = mean(weight), .groups = "drop") |>
    group_by(yr) |>
    summarise(
      n_spawners    = n_distinct(parent_pid),
      biom_spawn_kg = sum(parent_wt) / 1000,
      .groups = "drop"
    ) |>
    mutate(
      pcmax_val = factor(s$pc, levels = pcmax_levels),
      hab_type  = s$ht
    )
}) |>
  mutate(hab_type = factor(hab_type, levels = hab_levels))

spawn_yr_sum <- spawn_yr |>
  group_by(pcmax_val, hab_type) |>
  summarise(
    mean_spawners   = mean(n_spawners, na.rm = TRUE),
    sd_spawners     = sd(n_spawners, na.rm = TRUE),
    mean_spawn_biom = mean(biom_spawn_kg, na.rm = TRUE),
    sd_spawn_biom   = sd(biom_spawn_kg, na.rm = TRUE),
    .groups = "drop"
  )

spawn_part_sum <- spawn_yr |>
  left_join(pop_yr |> select(pcmax_val, hab_type, year, n),
            by = c("pcmax_val", "hab_type", "yr" = "year")) |>
  mutate(prop_spawn = n_spawners / n) |>
  group_by(pcmax_val, hab_type) |>
  summarise(
    mean_prop = mean(prop_spawn, na.rm = TRUE),
    sd_prop   = sd(prop_spawn,  na.rm = TRUE),
    .groups = "drop"
  )

11.1.2 Seasonal warm contribution

Code 
# Cold + Warm scenarios only
patch_seasonal <- purrr::map_dfr(
  ibm_list[sapply(ibm_list, \(s) s$ht == "Cold + Warm")],
  function(s) {
    s$df |>
      filter(date >= exp_start_date, !is.na(ggd)) |>
      mutate(
        daily_growth_g = weight * ggd,
        year   = year(date),
        season = factor(case_when(
          month(date) %in% c(12, 1, 2)  ~ "Winter",
          month(date) %in% c(3, 4, 5)   ~ "Spring",
          month(date) %in% c(6, 7, 8)   ~ "Summer",
          month(date) %in% c(9, 10, 11) ~ "Fall"
        ), levels = c("Winter", "Spring", "Summer", "Fall"))
      ) |>
      group_by(year, season, patch) |>
      summarise(total_g = sum(daily_growth_g, na.rm = TRUE), .groups = "drop") |>
      mutate(pcmax_val = factor(s$pc, levels = pcmax_levels))
  }
)

annual_total <- patch_seasonal |>
  mutate(pos_g = pmax(total_g, 0)) |>
  group_by(pcmax_val, year) |>
  summarise(annual_pos_g = sum(pos_g, na.rm = TRUE), .groups = "drop")

warm_season_sum <- patch_seasonal |>
  filter(patch == "warm") |>
  mutate(pos_g = pmax(total_g, 0)) |>
  left_join(annual_total, by = c("pcmax_val", "year")) |>
  mutate(f_warm = if_else(annual_pos_g > 0, pos_g / annual_pos_g, NA_real_)) |>
  group_by(pcmax_val, season) |>
  summarise(mean_f = mean(f_warm, na.rm = TRUE), sd_f = sd(f_warm, na.rm = TRUE), .groups = "drop")

# Annual warm share across all patches
warm_share <- patch_seasonal |>
  group_by(pcmax_val, year, patch) |>
  summarise(total_g = sum(total_g, na.rm = TRUE), .groups = "drop") |>
  pivot_wider(names_from = patch, values_from = total_g, values_fill = 0) |>
  mutate(
    pos_warm   = pmax(warm, 0),
    pos_cold   = pmax(cold, 0),
    f_warm_pos = if_else(pos_warm + pos_cold > 0,
                         pos_warm / (pos_warm + pos_cold),
                         NA_real_)
  ) |>
  group_by(pcmax_val) |>
  summarise(
    mean_f_warm = mean(f_warm_pos, na.rm = TRUE),
    sd_f_warm   = sd(f_warm_pos,   na.rm = TRUE),
    .groups     = "drop"
  )

dodge <- position_dodge(width = 0.75)

bar_plot <- function(data, y_var, y_sd, y_lab, pct = FALSE) {
  p <- ggplot(data, aes(x = pcmax_val, y = .data[[y_var]], fill = hab_type)) +
    geom_col(position = dodge, width = 0.65) +
    geom_errorbar(
      aes(ymin = pmax(0, .data[[y_var]] - .data[[y_sd]]),
          ymax = .data[[y_var]] + .data[[y_sd]]),
      position = dodge, width = 0.25, linewidth = 0.5
    ) +
    scale_fill_manual(values = hab_fills, name = "Habitat") +
    labs(x = "pcmax", y = y_lab) +
    theme_bw(base_size = 11) +
    theme(
      panel.grid.major.x = element_blank(),
      panel.grid.minor   = element_blank(),
      axis.text          = element_text(color = "black")
    )
  if (pct) {
    p + scale_y_continuous(labels = label_percent(accuracy = 1),
                           expand = expansion(mult = c(0, 0.08)))
  } else {
    p + scale_y_continuous(expand = expansion(mult = c(0, 0.08)))
  }
}

# Relative difference of means: (Cold + Warm - Cold only) / Cold only, by pcmax
rel_diff_plot <- function(data, mean_var, y_lab) {
  data |>
    select(pcmax_val, hab_type, value = all_of(mean_var)) |>
    pivot_wider(names_from = hab_type, values_from = value) |>
    mutate(rel = (`Cold + Warm` - `Cold only`) / `Cold only`) |>
    ggplot(aes(x = pcmax_val, y = rel)) +
    geom_col(width = 0.5) +
    scale_y_continuous(labels = label_percent(accuracy = 1),
                       expand = expansion(mult = c(0.08, 0.08))) +
    labs(x = "pcmax", y = y_lab) +
    theme_bw(base_size = 11) +
    theme(
      panel.grid.major.x = element_blank(),
      panel.grid.minor   = element_blank(),
      axis.text          = element_text(color = "black")
    )
}

11.4.2 Weekly biomass change by patch

Code 
patch_cols <- c("Cold patch" = "#6BAED6", "Warm patch" = "#FC8D59")
ref_year   <- year(sim_params$mindate) + 1L

week_abs <- purrr::map_dfr(
  ibm_list[sapply(ibm_list, \(s) s$ht == "Cold + Warm")],
  function(s) {
    s$df |>
      filter(date >= exp_start_date, !is.na(ggd), floor(age) >= 1) |>
      mutate(
        daily_growth_g = weight * ggd,
        year     = year(date),
        cal_week = week(date)
      ) |>
      group_by(year, cal_week, patch) |>
      summarise(
        mean_ggd = mean(ggd, na.rm = TRUE),
        total_g  = sum(daily_growth_g, na.rm = TRUE),
        .groups = "drop"
      ) |>
      mutate(pcmax_val = factor(s$pc, levels = pcmax_levels))
  }
) |>
  group_by(pcmax_val, cal_week, patch) |>
  summarise(
    mean_ggd  = mean(mean_ggd, na.rm = TRUE),
    mean_biom = mean(total_g,  na.rm = TRUE),
    .groups = "drop"
  ) |>
  mutate(
    patch_label = factor(if_else(patch == "cold", "Cold patch", "Warm patch"),
                         levels = c("Cold patch", "Warm patch")),
    week_date   = as.Date(paste0(ref_year, "-01-01")) + (cal_week - 1L) * 7L
  )

occ_week <- purrr::map_dfr(
  ibm_list[sapply(ibm_list, \(s) s$ht == "Cold + Warm")],
  function(s) {
    s$df |>
      filter(date >= exp_start_date, survived == 1, floor(age) >= 1) |>
      mutate(year = year(date), cal_week = week(date)) |>
      group_by(year, cal_week) |>
      summarise(prop_warm = mean(patch == "warm"), .groups = "drop") |>
      group_by(cal_week) |>
      summarise(mean_prop_warm = mean(prop_warm), .groups = "drop") |>
      mutate(
        pcmax_val = factor(s$pc, levels = pcmax_levels),
        week_date = as.Date(paste0(ref_year, "-01-01")) + (cal_week - 1L) * 7L
      )
  }
)

make_week_occ_plot <- function(pc) {
  wd    <- week_abs |> filter(pcmax_val == pc)
  od    <- occ_week |> filter(pcmax_val == pc)
  g_max <- max(wd$mean_biom, na.rm = TRUE)

  ggplot(wd, aes(x = week_date, y = mean_biom, fill = patch_label)) +
    geom_col(position = "stack", width = 7) +
    geom_hline(yintercept = 0, linewidth = 0.3) +
    geom_line(data = od,
              aes(x = week_date, y = mean_prop_warm * g_max),
              color = "black", linewidth = 0.7, inherit.aes = FALSE) +
    scale_fill_manual(values = patch_cols, name = "Patch") +
    scale_x_date(
      breaks = seq(as.Date(paste0(ref_year, "-01-01")),
                   as.Date(paste0(ref_year, "-12-01")), by = "month"),
      labels = \(x) substr(format(x, "%b"), 1, 1),
      expand = expansion(0)
    ) +
    scale_y_continuous(
      labels   = scales::label_comma(),
      sec.axis = sec_axis(~ . / g_max,
                          name   = "Prop. adults in warm patch",
                          labels = scales::label_percent(accuracy = 1))
    ) +
    labs(x = NULL, y = NULL, title = paste("pcmax =", pc)) +
    theme_bw(base_size = 12) +
    theme(
      panel.grid.minor   = element_blank(),
      axis.title.y.right = element_blank(),
      legend.position    = "right",
      axis.text          = element_text(color = "black")
    )
}

# Oct 1 census: experimental period, survivors only
exp_oct1 <- purrr::map_dfr(ibm_list, function(s) {
  s$df |>
    filter(date >= exp_start_date,
           month(date) == 10, day(date) == 1,
           survived == 1) |>
    mutate(
      pcmax_val = factor(s$pc, levels = pcmax_levels),
      hab_type  = s$ht,
      year      = year(date),
      age_class = floor(age)
    )
}) |>
  mutate(hab_type = factor(hab_type, levels = hab_levels))

pop_yr <- exp_oct1 |>
  group_by(pcmax_val, hab_type, year) |>
  summarise(n = n(), biomass_kg = sum(weight) / 1000, .groups = "drop")

pop_sum <- pop_yr |>
  group_by(pcmax_val, hab_type) |>
  summarise(mean_n = mean(n), sd_n = sd(n),
            mean_b = mean(biomass_kg), sd_b = sd(biomass_kg), .groups = "drop")

age0_sum <- exp_oct1 |>
  filter(age_class == 0) |>
  group_by(pcmax_val, hab_type, year) |>
  summarise(n_age0 = n(), .groups = "drop") |>
  group_by(pcmax_val, hab_type) |>
  summarise(mean_n0 = mean(n_age0), sd_n0 = sd(n_age0), .groups = "drop")

waa <- exp_oct1 |>
  filter(age_class >= 0, age_class <= 15) |>
  group_by(pcmax_val, hab_type, age_class) |>
  summarise(mean_wt = mean(weight), sd_wt = sd(weight), n = n(), .groups = "drop") |>
  filter(n >= 10)

spawn_yr <- purrr::map_dfr(ibm_list, function(s) {
  s$spawn_log |>
    mutate(date = sim_params$mindate + (dayofsim - 1L), yr = year(date)) |>
    filter(date >= exp_start_date) |>
    group_by(yr, parent_pid) |>
    summarize(parent_wt = mean(weight), .groups = "drop") |>
    group_by(yr) |>

      n_spawners    = n_distinct(parent_pid),
      biom_spawn_kg = sum(parent_wt) / 1000,

      pcmax_val = factor(s$pc, levels = pcmax_levels),
      hab_type  = s$ht

# Cold + Warm scenarios only
patch_seasonal <- purrr::map_dfr(
  ibm_list[sapply(ibm_list, \(s) s$ht == "Cold + Warm")],
  function(s) {
    s$df |>
      filter(date >= exp_start_date, !is.na(ggd)) |>
      mutate(
        daily_growth_g = weight * ggd,
        year   = year(date),
        season = factor(case_when(
          month(date) %in% c(12, 1, 2)  ~ "Winter",
          month(date) %in% c(3, 4, 5)   ~ "Spring",
          month(date) %in% c(6, 7, 8)   ~ "Summer",
          month(date) %in% c(9, 10, 11) ~ "Fall"
        ), levels = c("Winter", "Spring", "Summer", "Fall"))
      ) |>
      group_by(year, season, patch) |>
      summarise(total_g = sum(daily_growth_g, na.rm = TRUE), .groups = "drop") |>
      mutate(pcmax_val = factor(s$pc, levels = pcmax_levels))
  }
)

annual_total <- patch_seasonal |>
  mutate(pos_g = pmax(total_g, 0)) |>
  group_by(pcmax_val, year) |>
  summarise(annual_pos_g = sum(pos_g, na.rm = TRUE), .groups = "drop")

warm_season_sum <- patch_seasonal |>
  filter(patch == "warm") |>
  mutate(pos_g = pmax(total_g, 0)) |>
  left_join(annual_total, by = c("pcmax_val", "year")) |>
  mutate(f_warm = if_else(annual_pos_g > 0, pos_g / annual_pos_g, NA_real_)) |>
  group_by(pcmax_val, season) |>
  summarise(mean_f = mean(f_warm, na.rm = TRUE), sd_f = sd(f_warm, na.rm = TRUE), .groups = "drop")

# Annual warm share across all patches
warm_share <- patch_seasonal |>
  group_by(pcmax_val, year, patch) |>
  summarise(total_g = sum(total_g, na.rm = TRUE), .groups = "drop") |>
  pivot_wider(names_from = patch, values_from = total_g, values_fill = 0) |>
  mutate(
    pos_warm   = pmax(warm, 0),
    pos_cold   = pmax(cold, 0),
    f_warm_pos = if_else(pos_warm + pos_cold > 0,
                         pos_warm / (pos_warm + pos_cold),
                         NA_real_)
  ) |>
  group_by(pcmax_val) |>
  summarise(
    mean_f_warm = mean(f_warm_pos, na.rm = TRUE),
    sd_f_warm   = sd(f_warm_pos,   na.rm = TRUE),
    .groups     = "drop"
  )

dodge <- position_dodge(width = 0.75)

bar_plot <- function(data, y_var, y_sd, y_lab, pct = FALSE) {
  p <- ggplot(data, aes(x = pcmax_val, y = .data[[y_var]], fill = hab_type)) +
    geom_col(position = dodge, width = 0.65) +
    geom_errorbar(
      aes(ymin = pmax(0, .data[[y_var]] - .data[[y_sd]]),
          ymax = .data[[y_var]] + .data[[y_sd]]),
      position = dodge, width = 0.25, linewidth = 0.5
    ) +
    scale_fill_manual(values = hab_fills, name = "Habitat") +
    labs(x = "pcmax", y = y_lab) +
    theme_bw(base_size = 11) +
    theme(
      panel.grid.major.x = element_blank(),
      panel.grid.minor   = element_blank(),
      axis.text          = element_text(color = "black")
    )
  if (pct) {
    p + scale_y_continuous(labels = label_percent(accuracy = 1),
                           expand = expansion(mult = c(0, 0.08)))
  } else {
    p + scale_y_continuous(expand = expansion(mult = c(0, 0.08)))
  }
}

# Relative difference of means: (Cold + Warm - Cold only) / Cold only, by pcmax
rel_diff_plot <- function(data, mean_var, y_lab) {
  data |>
    select(pcmax_val, hab_type, value = all_of(mean_var)) |>
    pivot_wider(names_from = hab_type, values_from = value) |>
    mutate(rel = (`Cold + Warm` - `Cold only`) / `Cold only`) |>
    ggplot(aes(x = pcmax_val, y = rel)) +
    geom_col(width = 0.5) +
    scale_y_continuous(labels = label_percent(accuracy = 1),
                       expand = expansion(mult = c(0.08, 0.08))) +
    labs(x = "pcmax", y = y_lab) +
    theme_bw(base_size = 11) +
    theme(
      panel.grid.major.x = element_blank(),
      panel.grid.minor   = element_blank(),
      axis.text          = element_text(color = "black")
    )
}
