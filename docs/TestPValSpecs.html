patch_seasonal_mech <- purrr::map_dfr(
  ibm_list_mech[sapply(ibm_list_mech, \(s) s$ht == "Cold + Warm")],
  function(s) {
    s$df |>
      filter(date >= exp_start_date, !is.na(ggd)) |>
      mutate(
        daily_growth_g = weight * ggd,
        year   = year(date),
        season = factor(case_when(
          month(date) %in% c(12, 1, 2)  ~ "Winter",
          month(date) %in% c(3, 4, 5)   ~ "Spring",
          month(date) %in% c(6, 7, 8)   ~ "Summer",
          month(date) %in% c(9, 10, 11) ~ "Fall"
        ), levels = c("Winter", "Spring", "Summer", "Fall"))
      ) |>
      group_by(year, season, patch) |>
      summarise(total_g = sum(daily_growth_g, na.rm = TRUE), .groups = "drop") |>
      mutate(ptype = s$ptype)
  }
) |> mutate(ptype   = factor(ptype,   levels = ptype_levels))

annual_total_mech <- patch_seasonal_mech |>
  mutate(pos_g = pmax(total_g, 0)) |>
  group_by(ptype, year) |>
  summarise(annual_pos_g = sum(pos_g, na.rm = TRUE), .groups = "drop")

warm_season_sum <- patch_seasonal_mech |>
  filter(patch == "warm") |>
  mutate(pos_g = pmax(total_g, 0)) |>
  left_join(annual_total_mech, by = c("ptype", "year")) |>
  mutate(f_warm = if_else(annual_pos_g > 0, pos_g / annual_pos_g, NA_real_)) |>
  group_by(ptype, season) |>
  summarise(mean_f = mean(f_warm, na.rm = TRUE), sd_f = sd(f_warm, na.rm = TRUE), .groups = "drop")

# Theoretical p_bioe curves over temperature range (dd = 1, isolates temperature effect)
# Formula by p_calc type (from SimulationLoop.R):
#   legacy:    min(fT, pcmax)
#   corrected: min(fT, pcmax) / fT
#   fixed_p:   pcmax
p_bioe_curves <- expand_grid(
  temp    = seq(0, 25, by = 0.1),
  ptype   = factor(ptype_levels,   levels = ptype_levels)
) |>
  mutate(
    pcmax          = 0.5,
    fT             = fncTempDepend(temp),
    pcmax_adjusted = pmin(fT, pcmax),
    p_bioe = case_when(
      ptype == "legacy"    ~ pcmax_adjusted,
      ptype == "corrected" ~ pcmax_adjusted / pmax(fT, 1e-6),
      ptype == "fixedp"    ~ pcmax
    )
  )

# Reconstruct empirical p_bioe from sampled simulation days.
# p_bioe (= pcmax_adjusted_dd) is not stored in ibm_long, so we recompute it:
#   1. Sample days from the experimental period.
#   2. For each day × patch × age-class group, compute dominance-weighted effective
#      density via fncEffDensity(), then dd_factor = K / (K + eff_density).
#   3. Apply the p_calc formula to get per-fish p_bioe.
#   4. Summarise to daily patch means for plotting.
#
# Parameters matching the Cold+Warm STHD/CUTT scenarios:
A_patch        <- 1.0   # habitat area (A_cold = A_warm = 1)
K_patch        <- 400   # hyperbolic half-saturation constant (K_cold = K_warm = 400)
dominance_beta <- 1     # dominance exponent

set.seed(5317)
n_sample_days <- 50

p_bioe_empirical <- purrr::map_dfr(
  ibm_list_mech[sapply(ibm_list_mech, \(s) s$ht == "Cold + Warm")],
  function(s) {
    pcmax_val   <- 0.5
    ptype_val   <- s$ptype

    df           <- s$df |> filter(date >= exp_start_date, !is.na(ggd))
    sampled_days <- sample(unique(df$dayofsim), n_sample_days)

    df |>
      filter(dayofsim %in% sampled_days) |>
      # mutate(age_class = if_else(floor(age) == 0, "age0", "age1plus")) |>
      filter(floor(age) >= 1) |>
      group_by(dayofsim, patch) |>
      mutate(
        eff_density    = (fncEffDensity(weight, beta = dominance_beta) - 1) / A_patch,
        dd_factor      = K_patch / (K_patch + eff_density),
        fT             = fncTempDepend(temp),
        pcmax_adjusted = pmin(fT, pcmax_val),
        p_bioe = case_when(
          ptype_val == "legacy"    ~ pcmax_adjusted * dd_factor,
          ptype_val == "corrected" ~ pcmax_adjusted * dd_factor / pmax(fT, 1e-6),
          ptype_val == "fixedp"    ~ pcmax_val * dd_factor
        )
      ) |>
      # Collapse to daily patch mean (temp is identical across fish within a patch-day)
      # group_by(dayofsim, patch) |>
      # summarise(temp = first(temp), mean_p_bioe = mean(p_bioe), .groups = "drop") |>
      mutate(
        ptype       = factor(ptype_val,   levels = ptype_levels),
        patch_label = factor(if_else(patch == "cold", "Cold patch", "Warm patch"),
                             levels = c("Cold patch", "Warm patch"))
      )
  }
)

### Bioenergetics P by temperature

```{r fig.width=9, fig.height=5}

# Theoretical p_bioe curves over temperature range (dd = 1, isolates temperature effect)
# Formula by p_calc type (from SimulationLoop.R):
#   legacy:    min(fT, pcmax)
#   corrected: min(fT, pcmax) / fT
#   fixed_p:   pcmax
p_bioe_curves <- expand_grid(
  temp    = seq(0, 25, by = 0.1),
  ptype   = factor(ptype_levels,   levels = ptype_levels)
) |>
  mutate(
    pcmax          = 0.5,
    fT             = fncTempDepend(temp),
    pcmax_adjusted = pmin(fT, pcmax),
    p_bioe = case_when(
      ptype == "legacy"    ~ pcmax_adjusted,
      ptype == "corrected" ~ pcmax_adjusted / pmax(fT, 1e-6),
      ptype == "fixedp"    ~ pcmax
    )
  )

# Reconstruct empirical p_bioe from sampled simulation days.
# p_bioe (= pcmax_adjusted_dd) is not stored in ibm_long, so we recompute it:
#   1. Sample days from the experimental period.
#   2. For each day × patch × age-class group, compute dominance-weighted effective
#      density via fncEffDensity(), then dd_factor = K / (K + eff_density).
#   3. Apply the p_calc formula to get per-fish p_bioe.
#   4. Summarise to daily patch means for plotting.
#
# Parameters matching the Cold+Warm STHD/CUTT scenarios:
A_patch        <- 1.0   # habitat area (A_cold = A_warm = 1)
K_patch        <- 400   # hyperbolic half-saturation constant (K_cold = K_warm = 400)
dominance_beta <- 1     # dominance exponent

set.seed(5317)
n_sample_days <- 50

p_bioe_empirical <- purrr::map_dfr(
  ibm_list_mech[sapply(ibm_list_mech, \(s) s$ht == "Cold + Warm")],
  function(s) {
    pcmax_val   <- 0.5
    ptype_val   <- s$ptype

    df           <- s$df |> filter(date >= exp_start_date, !is.na(ggd))
    sampled_days <- sample(unique(df$dayofsim), n_sample_days)

    df |>
      filter(dayofsim %in% sampled_days) |>
      # mutate(age_class = if_else(floor(age) == 0, "age0", "age1plus")) |>
      filter(floor(age) >= 1) |>
      group_by(dayofsim, patch) |>
      mutate(
        eff_density    = (fncEffDensity(weight, beta = dominance_beta) - 1) / A_patch,
        dd_factor      = K_patch / (K_patch + eff_density),
        fT             = fncTempDepend(temp),
        pcmax_adjusted = pmin(fT, pcmax_val),
        p_bioe = case_when(
          ptype_val == "legacy"    ~ pcmax_adjusted * dd_factor,
          ptype_val == "corrected" ~ pcmax_adjusted * dd_factor / pmax(fT, 1e-6),
          ptype_val == "fixedp"    ~ pcmax_val * dd_factor
        )
      ) |>
      # Collapse to daily patch mean (temp is identical across fish within a patch-day)
      # group_by(dayofsim, patch) |>
      # summarise(temp = first(temp), mean_p_bioe = mean(p_bioe), .groups = "drop") |>
      mutate(
        ptype       = factor(ptype_val,   levels = ptype_levels),
        patch_label = factor(if_else(patch == "cold", "Cold patch", "Warm patch"),

  }
```
