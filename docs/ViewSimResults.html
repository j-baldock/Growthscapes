13.1 Setup

Load each result .rds object and bind directly into a list. Using a helper function ensures each res object is freed from memory as soon as the objects needed for plotting (ibm_summary, spawn_log, fish_registry, habitat_df) are extracted. The large ibm_long is deliberately not kept.

Code 
# Extract params from one file (small object)
res            <- readRDS("results/Baseline_ColdOnly_005percold_rep01.rds")
sim_params     <- res$params
habitat_df     <- res$habitat_df
exp_start_date <- sim_params$mindate + years(sim_params$nyears_burnin)
rm(res)

# Helper: read one scenario, keep only the objects needed by plot_population(),
# and free res (including the large ibm_long) before returning
load_scen <- function(path, pc, ht, pcmax_cold, pcmax_warm) {
  res <- readRDS(path)
  list(
    ibm_summary   = res$ibm_summary,
    spawn_log     = res$spawn_log,
    fish_registry = res$fish_registry,
    habitat_df    = res$habitat_df,
    pc = pc, ht = ht, pcmax_cold = pcmax_cold, pcmax_warm = pcmax_warm
  )
}

ibm_list_mech <- list(
